// fxdock.exe - pembantu Dock macOS untuk Windows (dikompilasi dengan csc.exe bawaan Windows, C# 5).
// JANGAN pakai sintaks baru (?., $"", out var, dll).
//   fxdock.exe watch <pidInduk> [fileCatatan] -> pantau jendela aplikasi (satu JSON per baris), terima perintah lewat stdin:
//                                   focus <hwnd> | min <hwnd> | close <hwnd>
//   fxdock.exe tb get            -> cetak 1 kalau taskbar auto-hide, 0 kalau tidak
//   fxdock.exe tb set <0|1>      -> atur auto-hide taskbar, lalu cetak status terbaru
//   fxdock.exe tb vis <0|1>      -> sembunyikan (0) / tampilkan (1) jendela taskbar sepenuhnya

    // JSON string; karakter non-ASCII di-escape supaya aman dari masalah encoding stdout.
    static string J(string s)
    {
        if (s == null) return "\"\"";
        StringBuilder sb = new StringBuilder();
        sb.Append('"');
        foreach (char c in s)
        {
            if (c == '"') sb.Append("\\\"");
            else if (c == '\\') sb.Append("\\\\");
            else if (c == '\n') sb.Append("\\n");
            else if (c == '\r') sb.Append("\\r");
            else if (c == '\t') sb.Append("\\t");
            else if (c < 32 || c > 126) sb.Append("\\u").Append(((int)c).ToString("x4"));
            else sb.Append(c);
        }
        sb.Append('"');
        return sb.ToString();
    }

    static string ClassOf(IntPtr h)
    {
        StringBuilder sb = new StringBuilder(160);
        GetClassName(h, sb, 160);
        return sb.ToString();
    }

    static string ProcPath(uint pid)
    {
        string v;
        if (pathCache.TryGetValue(pid, out v)) return v;
        v = "";
        IntPtr hp = IntPtr.Zero;
        try
        {
            hp = OpenProcess(0x1000, false, pid); // PROCESS_QUERY_LIMITED_INFORMATION
            if (hp != IntPtr.Zero)
            {
                StringBuilder sb = new StringBuilder(1024);
                int n = sb.Capacity;
                if (QueryFullProcessImageName(hp, 0, sb, ref n)) v = sb.ToString();
            }
        }
        catch (Exception) { }
        finally { if (hp != IntPtr.Zero) CloseHandle(hp); }
        pathCache[pid] = v;
        return v;
    }

    // AUMID hanya dipakai untuk aplikasi Store (path berisi \WindowsApps\), supaya ikon dan pin cocok.
    static string AumidOf(uint pid, string path)
    {
        if (path.IndexOf("\\WindowsApps\\", StringComparison.OrdinalIgnoreCase) < 0) return "";
        string v;
        if (aumidCache.TryGetValue(pid, out v)) return v;
        v = "";
        IntPtr hp = IntPtr.Zero;
        try
        {
            hp = OpenProcess(0x1000, false, pid);
            if (hp != IntPtr.Zero)
            {
                int n = 300;
                StringBuilder sb = new StringBuilder(n);
                int rc = GetApplicationUserModelId(hp, ref n, sb);
                if (rc == 0) v = sb.ToString();
            }
        }
        catch (Exception) { }
        finally { if (hp != IntPtr.Zero) CloseHandle(hp); }
        aumidCache[pid] = v;
        return v;
    }

    // Jendela aplikasi Store dibungkus ApplicationFrameHost; proses sebenarnya ada di jendela anak CoreWindow.
    static uint RealPid(IntPtr h, uint pid)
    {
        string p = ProcPath(pid);
        if (p.ToLowerInvariant().EndsWith("applicationframehost.exe"))
        {
            uint found = 0;
            EnumChildWindows(h, delegate (IntPtr c, IntPtr l)
            {
                uint cp;
                GetWindowThreadProcessId(c, out cp);
                if (cp != pid && cp != 0) { found = cp; return false; }
                return true;
            }, IntPtr.Zero);
            if (found != 0) return found;
        }
        return pid;
    }

    static bool Eligible(IntPtr h)
    {
        if (!IsWindowVisible(h)) return false;
        int ck = 0;
        try { DwmGetWindowAttribute(h, 14, out ck, 4); } catch (Exception) { ck = 0; }
        if (ck != 0 && !IsIconic(h)) return false;
        int ex = GetWindowLong(h, -20);
        bool appw = (ex & 0x40000) != 0;          // WS_EX_APPWINDOW
        if ((ex & 0x80) != 0 && !appw) return false;        // WS_EX_TOOLWINDOW
        if ((ex & 0x08000000) != 0 && !appw) return false;  // WS_EX_NOACTIVATE (jendela overlay kita sendiri)
        if (GetWindow(h, 4) != IntPtr.Zero && !appw) return false; // punya owner = dialog
        if (GetWindowTextLength(h) <= 0) return false;
        string cls = ClassOf(h);
        if (Array.IndexOf(skipCls, cls) >= 0) return false;
        if (!IsIconic(h))
        {
            RECT r;
            if (GetWindowRect(h, out r))
            {
                if (r.R - r.L < 80 || r.B - r.T < 40) return false;
            }
        }
        return true;
    }

    // true kalau jendela aktif menutupi seluruh monitornya (game / video layar penuh)
    static bool IsFullscreen(IntPtr h)
    {
        if (h == IntPtr.Zero || !IsWindowVisible(h) || IsIconic(h)) return false;
        string cls = ClassOf(h);
        if (Array.IndexOf(skipCls, cls) >= 0) return false;
        if ((GetWindowLong(h, -16) & 0xC00000) == 0xC00000) return false;   // jendela biasa yang di-maximize (punya title bar) bukan layar penuh
        RECT r;
        if (!GetWindowRect(h, out r)) return false;
        IntPtr mon = MonitorFromWindow(h, 2);
        if (mon == IntPtr.Zero) return false;
        MONITORINFO mi = new MONITORINFO();
        mi.cbSize = Marshal.SizeOf(typeof(MONITORINFO));
        if (!GetMonitorInfo(mon, ref mi)) return false;
        return r.L <= mi.rcMonitor.L + 1 && r.T <= mi.rcMonitor.T + 1 && r.R >= mi.rcMonitor.R - 1 && r.B >= mi.rcMonitor.B - 1;
    }

    static string Snapshot()
    {
        List<Win> list = new List<Win>();
        EnumWindows(delegate (IntPtr h, IntPtr l)
        {
            try
            {
                if (!Eligible(h)) return true;
                uint pid;
                GetWindowThreadProcessId(h, out pid);
                if (pid == 0) return true;
                uint rp = RealPid(h, pid);
                string path = ProcPath(rp);
                string exeName = path.Length > 0 ? path.Substring(path.LastIndexOf('\\') + 1).ToLowerInvariant() : "";
                if (exeName.Length == 0)
                {
                    // proses lebih tinggi (admin): path tidak terbaca, pakai nama proses saja
                    try { exeName = Process.GetProcessById((int)rp).ProcessName.ToLowerInvariant() + ".exe"; } catch (Exception) { exeName = ""; }
                }
                if (exeName.Length == 0) return true;
                if (Array.IndexOf(skipExe, exeName) >= 0) return true;
                Win w = new Win();
                w.H = h.ToInt64();
                w.Pid = rp;
                w.Exe = path.Length > 0 ? path : exeName;
                w.Aumid = AumidOf(rp, path);
                w.Min = IsIconic(h);
                if (withTitle) { StringBuilder tb = new StringBuilder(260); GetWindowText(h, tb, 260); w.Title = tb.ToString(); }
                list.Add(w);
            }
            catch (Exception) { }
            return true;
        }, IntPtr.Zero);

        // buang cache proses yang sudah tidak ada
        if (pathCache.Count > 300 || aumidCache.Count > 300)
        {
            pathCache.Clear();
            aumidCache.Clear();
        }

        IntPtr fg = GetForegroundWindow();
        bool fs = false;
        try { fs = IsFullscreen(fg); } catch (Exception) { fs = false; }

        StringBuilder sb = new StringBuilder();
        sb.Append("{\"t\":\"wins\",\"fg\":").Append(fg.ToInt64()).Append(",\"fs\":").Append(fs ? 1 : 0).Append(",\"l\":[");
        for (int i = 0; i < list.Count; i++)
        {
            Win w = list[i];
            if (i > 0) sb.Append(',');
            sb.Append("{\"h\":").Append(w.H)
              .Append(",\"p\":").Append(w.Pid)
              .Append(",\"x\":").Append(J(w.Exe))
              .Append(",\"a\":").Append(J(w.Aumid))
              .Append(",\"m\":").Append(w.Min ? 1 : 0);
            if (withTitle) sb.Append(",\"t\":").Append(J(w.Title));
            sb.Append('}');
        }
        sb.Append("]}");
        return sb.ToString();
    }

    static void Focus(IntPtr h)
    {
        if (h == IntPtr.Zero || !IsWindow(h)) return;
        if (IsIconic(h)) ShowWindow(h, 9); // SW_RESTORE
        uint me = GetCurrentThreadId();
        IntPtr fg = GetForegroundWindow();
        uint fpid;
        uint fgT = fg == IntPtr.Zero ? 0 : GetWindowThreadProcessId(fg, out fpid);
        bool att = false;
        if (fgT != 0 && fgT != me) att = AttachThreadInput(me, fgT, true);
        BringWindowToTop(h);
        ShowWindow(h, 5); // SW_SHOW
        SetForegroundWindow(h);
        if (att) AttachThreadInput(me, fgT, false);
        if (GetForegroundWindow() != h)
        {
            // cadangan: ketukan tombol Alt melepas kunci latar depan Windows
            keybd_event(0x12, 0, 0, UIntPtr.Zero);
            keybd_event(0x12, 0, 2, UIntPtr.Zero);
            BringWindowToTop(h);
            SetForegroundWindow(h);
        }
    }

    // Pulihkan (kalau diminimize) lalu bawa ke depan. SwitchToThisWindow menembus kunci latar depan Windows.
    static void Activate(IntPtr h)
    {
        if (h == IntPtr.Zero || !IsWindow(h)) return;
        if (IsIconic(h)) PostMessage(h, 0x0112, new IntPtr(0xF120), IntPtr.Zero); // WM_SYSCOMMAND SC_RESTORE
        Focus(h);
        try { SwitchToThisWindow(h, true); } catch (Exception) { }
    }

    // Cari jendela milik exe tertentu sampai ke yang tidak masuk daftar Dock (diminimize / cloaked), lalu tampilkan.
    static void ShowExe(string name)
    {
        string want = name.ToLowerInvariant();
        IntPtr best = IntPtr.Zero;
        int bestScore = -1;
        EnumWindows(delegate (IntPtr h, IntPtr l)
        {
            try
            {
                if (!IsWindowVisible(h)) return true;
                if (GetWindow(h, 4) != IntPtr.Zero) return true;
                if (GetWindowTextLength(h) <= 0) return true;
                if (Array.IndexOf(skipCls, ClassOf(h)) >= 0) return true;
                int ex = GetWindowLong(h, -20);
                if ((ex & 0x80) != 0 && (ex & 0x40000) == 0) return true;
                uint pid;
                GetWindowThreadProcessId(h, out pid);
                if (pid == 0) return true;
                string pth = ProcPath(RealPid(h, pid)).ToLowerInvariant();
                string b = pth.Length > 0 ? pth.Substring(pth.LastIndexOf('\\') + 1) : "";
                if (b != want) return true;
                int score = IsIconic(h) ? 2 : 1;
                if (score > bestScore) { bestScore = score; best = h; }
            }
            catch (Exception) { }
            return true;
        }, IntPtr.Zero);
        if (best != IntPtr.Zero) Activate(best);
        Emit("{\"t\":\"shown\",\"x\":" + J(name) + ",\"ok\":" + (best != IntPtr.Zero ? 1 : 0) + "}");
    }

    static void Cmd(string line)
    {
        if (string.IsNullOrEmpty(line)) return;
        if (line == "start")
        {
            keybd_event(0x5B, 0, 0, UIntPtr.Zero); // tombol Windows: buka menu Start
            keybd_event(0x5B, 0, 2, UIntPtr.Zero);
            return;
        }
        if (line.StartsWith("show "))
        {
            try { ShowExe(line.Substring(5).Trim()); } catch (Exception) { Emit("{\"t\":\"shown\",\"ok\":0}"); }
            return;
        }
        string[] p = line.Split(' ');
        if (p.Length < 2) return;
        long hv;
        if (!long.TryParse(p[1], out hv) || hv == 0) return;
        IntPtr h = new IntPtr(hv);
        try
        {
            if (p[0] == "focus") Activate(h);
            else if (p[0] == "min") { if (IsWindow(h)) ShowWindow(h, 6); }       // SW_MINIMIZE
            else if (p[0] == "close") { if (IsWindow(h)) PostMessage(h, 0x0010, IntPtr.Zero, IntPtr.Zero); } // WM_CLOSE
            else if (p[0] == "max") { if (IsWindow(h)) ShowWindow(h, 3); }       // SW_MAXIMIZE
            else if (p[0] == "place" && p.Length >= 6)
            {
                int px, py, pw, ph;
                if (IsWindow(h) && int.TryParse(p[2], out px) && int.TryParse(p[3], out py) && int.TryParse(p[4], out pw) && int.TryParse(p[5], out ph) && pw > 50 && ph > 50)
                {
                    ShowWindow(h, 9);                                              // SW_RESTORE (kalau sedang maximize)
                    SetWindowPos(h, IntPtr.Zero, px, py, pw, ph, 0x0014);          // SWP_NOZORDER | SWP_NOACTIVATE
                }
            }
        }
        catch (Exception) { }
    }

    // ----- taskbar auto-hide (ABM_GETSTATE = 4, ABM_SETSTATE = 10, ABS_AUTOHIDE = 1)
    static APPBARDATA Abd()
    {
        APPBARDATA d = new APPBARDATA();
        d.cbSize = (uint)Marshal.SizeOf(typeof(APPBARDATA));
        d.hWnd = FindWindow("Shell_TrayWnd", null);
        return d;
    }

    static int TbGet()
    {
        APPBARDATA d = Abd();
        long st = (long)SHAppBarMessage(4, ref d).ToUInt64();
        return (st & 1) != 0 ? 1 : 0;
    }

    static void TbSet(int on)
    {
        APPBARDATA d = Abd();
        long st = (long)SHAppBarMessage(4, ref d).ToUInt64();
        long want = on != 0 ? (st | 1) : (st & ~1L);
        d.lParam = new IntPtr(want);
        SHAppBarMessage(10, ref d);
    }

    // Menyembunyikan jendela taskbar sepenuhnya (tidak muncul walau kursor ke tepi layar)
    static void TbVis(int on)
    {
        int cmd = on != 0 ? 5 : 0; // SW_SHOW / SW_HIDE
        IntPtr h = FindWindow("Shell_TrayWnd", null);
        if (h != IntPtr.Zero) ShowWindow(h, cmd);
        EnumWindows(delegate (IntPtr w, IntPtr l)
        {
            if (ClassOf(w) == "Shell_SecondaryTrayWnd") ShowWindow(w, cmd);
            return true;
        }, IntPtr.Zero);
    }

    // ----- ikon desktop (halaman utama): sama seperti klik kanan desktop > Tampilan > Tampilkan ikon desktop
    static IntPtr DeskDefView()
    {
        IntPtr defv = IntPtr.Zero;
        IntPtr prog = FindWindow("Progman", null);
        if (prog != IntPtr.Zero) defv = FindWindowEx(prog, IntPtr.Zero, "SHELLDLL_DefView", null);
        if (defv == IntPtr.Zero)
        {
            IntPtr found = IntPtr.Zero;
            EnumWindows(delegate (IntPtr w, IntPtr l)
            {
                if (ClassOf(w) == "WorkerW")
                {
                    IntPtr d = FindWindowEx(w, IntPtr.Zero, "SHELLDLL_DefView", null);
                    if (d != IntPtr.Zero) { found = d; return false; }
                }
                return true;
            }, IntPtr.Zero);
            defv = found;
        }
        return defv;
    }

    static bool IconsVisible()
    {
        IntPtr defv = DeskDefView();
        if (defv == IntPtr.Zero) return true;
        IntPtr lv = FindWindowEx(defv, IntPtr.Zero, "SysListView32", null);
        if (lv == IntPtr.Zero) return true;
        return IsWindowVisible(lv);
    }

    static void IconsSet(bool show)
    {
        if (IconsVisible() == show) return;
        IntPtr defv = DeskDefView();
        if (defv == IntPtr.Zero) return;
        SendMessage(defv, 0x0111, new IntPtr(0x7402), IntPtr.Zero); // WM_COMMAND 0x7402 = bolak-balik tampil/sembunyi ikon desktop
    }

    // Kalau aplikasi induk mati, ikon desktop yang kita sembunyikan dimunculkan lagi (file catatan ada = kita yang menyembunyikan).
    static string iconFlag = "";
    static void RestoreIcons()
    {
        try
        {
            if (iconFlag.Length == 0 || !System.IO.File.Exists(iconFlag)) return;
            IconsSet(true);
            System.IO.File.Delete(iconFlag);
        }
        catch (Exception) { }
    }

    // Kalau aplikasi induk mati, taskbar yang kita sembunyikan dikembalikan (file catatan berisi status aslinya).
    static string flagFile = "";
    static void RestoreTb()
    {
        try
        {
            if (flagFile.Length == 0 || !System.IO.File.Exists(flagFile)) return;
            string t = System.IO.File.ReadAllText(flagFile);
            int i = t.IndexOf("orig", StringComparison.Ordinal);
            int orig = -1;
            if (i >= 0)
            {
                int k = t.IndexOf(':', i);
                if (k >= 0)
                {
                    for (int j = k + 1; j < t.Length; j++)
                    {
                        if (t[j] == '0') { orig = 0; break; }
                        if (t[j] == '1') { orig = 1; break; }
                    }
                }
            }
            if (orig < 0) return;
            TbVis(1);
            TbSet(orig);
        }
        catch (Exception) { }
    }

    static int Main(string[] a)
    {
        try { SetProcessDpiAwarenessContext(new IntPtr(-4)); }
        catch (Exception)
        {
            try { SetProcessDPIAware(); } catch (Exception) { }
        }
        if (a.Length < 1) return 1;

        if (a[0] == "tb")
        {
            try
            {
                if (a.Length >= 3 && a[1] == "set")
                {
                    TbSet(a[2] == "1" ? 1 : 0);
                    Thread.Sleep(250);
                }
                else if (a.Length >= 3 && a[1] == "vis")
                {
                    TbVis(a[2] == "1" ? 1 : 0);
                    Thread.Sleep(150);
                }
                Console.Out.WriteLine(TbGet().ToString());
                Console.Out.Flush();
                return 0;
            }

    public static void Emit(string json)
    {
        lock (O)

            Console.Out.Flush();
        }
    }

    // JSON string, semua karakter non-ASCII di-escape supaya aman dari masalah encoding stdout.
    public static string J(string s)
    {
        if (s == null) return "\"\"";
        StringBuilder sb = new StringBuilder();
        sb.Append('"');
        foreach (char c in s)
        {
            if (c == '"') sb.Append("\\\"");
            else if (c == '\\') sb.Append("\\\\");
            else if (c == '\n') sb.Append("\\n");
            else if (c == '\r') sb.Append("\\r");
            else if (c == '\t') sb.Append("\\t");
            else if (c < 32 || c > 126) sb.Append("\\u").Append(((int)c).ToString("x4"));
            else sb.Append(c);
        }
        sb.Append('"');
        return sb.ToString();
    }

    [DllImport("user32.dll")] static extern bool SetProcessDPIAware();
    [DllImport("user32.dll")] static extern bool SetProcessDpiAwarenessContext(IntPtr ctx);

    [STAThread]

    {

        try

        }

    }

static class IconUtil
{
    [ComImport, Guid("bcc18b79-ba16-442f-80c4-8a59c30c463b"), InterfaceType(ComInterfaceType.InterfaceIsIUnknown)]
    interface IShellItemImageFactory
    {
        [PreserveSig] int GetImage(SIZE size, int flags, out IntPtr phbm);
    }

    [StructLayout(LayoutKind.Sequential)]
    struct SIZE { public int cx; public int cy; }

    [DllImport("shell32.dll", CharSet = CharSet.Unicode, PreserveSig = false)]
    static extern void SHCreateItemFromParsingName(
        [In, MarshalAs(UnmanagedType.LPWStr)] string path, IntPtr pbc,
        [In, MarshalAs(UnmanagedType.LPStruct)] Guid riid,
        [Out, MarshalAs(UnmanagedType.Interface)] out IShellItemImageFactory ppv);

    [DllImport("gdi32.dll")] static extern bool DeleteObject(IntPtr h);

    static readonly Guid IID = new Guid("bcc18b79-ba16-442f-80c4-8a59c30c463b");

    // id: path file .exe ATAU AUMID (aplikasi Store / terdaftar)
    public static byte[] Png(string id, int px)
    {
        if (string.IsNullOrEmpty(id)) return null;
        if (px < 16) px = 16;
        if (px > 256) px = 256;
        string p = File.Exists(id) ? id : "shell:AppsFolder\\" + id;
        IShellItemImageFactory f = null;
        IntPtr hbm = IntPtr.Zero;
        try
        {
            SHCreateItemFromParsingName(p, IntPtr.Zero, IID, out f);
            SIZE s; s.cx = px; s.cy = px;
            int hr = f.GetImage(s, 4 /* SIIGBF_ICONONLY */, out hbm);
            if (hr != 0 || hbm == IntPtr.Zero) return null;
            using (Bitmap src = Image.FromHbitmap(hbm))
            {
                BitmapData d = src.LockBits(new Rectangle(0, 0, src.Width, src.Height), ImageLockMode.ReadOnly, src.PixelFormat);
                try
                {
                    if (Image.GetPixelFormatSize(src.PixelFormat) < 32)
                    {
                        using (MemoryStream m0 = new MemoryStream()) { src.Save(m0, ImageFormat.Png); return m0.ToArray(); }
                    }
                    using (Bitmap argb = new Bitmap(d.Width, d.Height, d.Stride, PixelFormat.Format32bppArgb, d.Scan0))
                    using (MemoryStream m = new MemoryStream())
                    {
                        argb.Save(m, ImageFormat.Png);
                        return m.ToArray();
                    }
                }
                finally { src.UnlockBits(d); }
            }
        }
        catch (Exception) { return null; }
        finally
        {
            if (hbm != IntPtr.Zero) DeleteObject(hbm);
            if (f != null) { try { Marshal.ReleaseComObject(f); } catch (Exception) { } }
        }
    }

    static long Sig(string f)
    {
        try
        {
            FileInfo fi = new FileInfo(f);

        catch (Exception) { return 0; }

    static void CopyShared(string src, string dst)
    {
        if (!File.Exists(src))
        {
            try { if (File.Exists(dst)) File.Delete(dst); } catch (Exception) { }
            return;
        }
        using (FileStream i = new FileStream(src, FileMode.Open, FileAccess.Read, FileShare.ReadWrite | FileShare.Delete))
        using (FileStream o = new FileStream(dst, FileMode.Create, FileAccess.Write, FileShare.None))
        {
            byte[] buf = new byte[81920];
            int n;
            while ((n = i.Read(buf, 0, buf.Length)) > 0) o.Write(buf, 0, n);
        }

    static List<Row> Query(string file, long after, bool maxOnly, out long maxId)
    {
        List<Row> rows = new List<Row>();
        maxId = after;
        IntPtr db = IntPtr.Zero, st = IntPtr.Zero;
        if (Open(file, out db) != 0) { if (db != IntPtr.Zero) Close(db); throw new Exception("Gagal membuka salinan database notifikasi"); }
        try
        {
            string sql = maxOnly
                ? "SELECT IFNULL(MAX(Id),0) FROM Notification"
                : "SELECT n.Id, IFNULL(h.PrimaryId,''), n.Payload FROM Notification n LEFT JOIN NotificationHandler h ON h.RecordId = n.HandlerId WHERE n.Type = 'toast' AND n.Id > ?1 ORDER BY n.Id LIMIT 20";
            if (Prepare(db, sql, -1, out st, IntPtr.Zero) != 0) throw new Exception("Gagal membaca tabel notifikasi");
            if (!maxOnly) BindInt64(st, 1, after);
            while (Step(st) == 100 /* SQLITE_ROW */)
            {
                if (maxOnly) { maxId = ColInt64(st, 0); continue; }
                Row r = new Row();
                r.Id = ColInt64(st, 0);
                IntPtr tp = ColText16(st, 1);
                r.App = tp == IntPtr.Zero ? "" : Marshal.PtrToStringUni(tp);
                IntPtr bp = ColBlob(st, 2);
                int bn = ColBytes(st, 2);
                if (bp != IntPtr.Zero && bn > 0)
                {
                    r.Payload = new byte[bn];
                    Marshal.Copy(bp, r.Payload, 0, bn);
                }
                rows.Add(r);
                if (r.Id > maxId) maxId = r.Id;
            }
        }
        finally
        {
            if (st != IntPtr.Zero) Finalize(st);
            Close(db);
        }
        return rows;
    }

    static void Handle(Row r)
    {
        if (r.Payload == null) return;
        string xml = Encoding.UTF8.GetString(r.Payload).Trim('\0', '﻿', ' ', '\r', '\n', '\t');
        if (xml.Length == 0) return;
        XmlDocument doc = new XmlDocument();
        try { doc.LoadXml(xml); } catch (Exception) { return; }
        XmlNodeList texts = doc.SelectNodes("//binding/text");
        List<string> list = new List<string>();
        if (texts != null)
        {
            foreach (XmlNode t in texts)
            {
                XmlAttribute pl = t.Attributes == null ? null : t.Attributes["placement"];
                if (pl != null && pl.Value == "attribution") continue;
                string v = (t.InnerText ?? "").Trim();
                if (v.Length > 0) list.Add(v);
            }
        }
        if (list.Count == 0) return;
        string title = list[0];
        StringBuilder body = new StringBuilder();
        for (int i = 1; i < list.Count; i++)
        {
            if (body.Length > 0) body.Append('\n');
            body.Append(list[i]);
        }
        string body2 = body.ToString();
        if (body2.Length > 400) body2 = body2.Substring(0, 400);
        if (title.Length > 200) title = title.Substring(0, 200);
        string img = "";
        XmlNode im = doc.SelectSingleNode("//binding/image[@placement='appLogoOverride']");
        if (im != null && im.Attributes != null && im.Attributes["src"] != null) img = im.Attributes["src"].Value;
        Program.Emit("{\"t\":\"toast\",\"id\":" + r.Id + ",\"app\":" + Program.J(r.App) + ",\"title\":" + Program.J(title)
            + ",\"body\":" + Program.J(body2) + ",\"img\":" + Program.J(img) + "}");
    }

    public static void Run()
    {
        string local = Environment.GetEnvironmentVariable("LOCALAPPDATA");
        string dir = Path.Combine(local ?? "", "Microsoft\\Windows\\Notifications");
        string db = Path.Combine(dir, "wpndatabase.db");

        {
            Program.Emit("{\"t\":\"err\",\"m\":\"Database notifikasi Windows tidak ditemukan\"}");
            Thread.Sleep(Timeout.Infinite);
            return;
        }
        string tmp = Path.Combine(Path.GetTempPath(), "fxn" + Process.GetCurrentProcess().Id);
        Directory.CreateDirectory(tmp);
        string tdb = Path.Combine(tmp, "w.db");

        string lastErr = "";
        while (true)

                if (s != sig || last < 0)

                    CopyShared(db + "-wal", tdb + "-wal");
                    CopyShared(db + "-shm", tdb + "-shm");
                    long mx;
                    if (last < 0)
                    {
                        Query(tdb, 0, true, out mx);
                        last = mx;
                        Program.Emit("{\"t\":\"ready\"}");
                        lastErr = "";

                    else
                    {

                        foreach (Row r in rows)
                        {
                            try { Handle(r); } catch (Exception) { }
                        }
                        if (mx > last) last = mx;
                        lastErr = "";
                    }
                }
            }
            catch (Exception e)
            {

                if (m != lastErr)
                {
                    lastErr = m;
                    Program.Emit("{\"t\":\"err\",\"m\":" + Program.J(m) + "}");
                }
                sig = 0;
            }

    static string[] ProcInfo(uint pid)
    {
        string[] r;
        if (procCache.TryGetValue(pid, out r)) return r;
        string exe = "", aumid = "";
        IntPtr hp = OpenProcess(0x1000 /* PROCESS_QUERY_LIMITED_INFORMATION */, false, pid);
        if (hp != IntPtr.Zero)
        {
            try
            {
                StringBuilder sb = new StringBuilder(1024);
                int n = sb.Capacity;
                if (QueryFullProcessImageName(hp, 0, sb, ref n)) exe = sb.ToString();
                StringBuilder ab = new StringBuilder(256);

                if (GetApplicationUserModelId(hp, ref an, ab) == 0) aumid = ab.ToString();
            }
            catch (Exception) { }
            finally { CloseHandle(hp); }
        }

        if (procCache.Count > 400) procCache.Clear();

    // 0 = tidak tersembunyi, 1 = tersembunyi oleh aplikasi (dianggap ditutup), 2 = disembunyikan desktop virtual
    static int CloakKind(IntPtr h)
    {
        int c = 0;
        try { DwmGetWindowAttribute(h, 14, out c, 4); } catch (Exception) { return 0; }
        if (c == 0) return 0;
        if ((c & 2) != 0) return 2;
        return 1;
    }

    static bool Candidate(IntPtr h, uint own, out Info info, out int hidden)
    {
        info = null;
        hidden = 0;
        if (!IsWindowVisible(h)) { hidden = 1; return false; }
        int ck = CloakKind(h);
        if (ck != 0) { hidden = ck == 2 ? 2 : 1; return false; }
        int ex = GetWindowLong(h, -20);
        bool appw = (ex & 0x40000) != 0;
        if ((ex & 0x80) != 0 && !appw) return false;
        if ((ex & 0x08000000) != 0 && !appw) return false;
        if (GetWindow(h, 4) != IntPtr.Zero) return false;
        if (GetWindowTextLength(h) <= 0) return false;
        string cls = ClassOf(h);
        if (skipClass.Contains(cls)) return false;
        uint pid;
        GetWindowThreadProcessId(h, out pid);
        if (pid == own) return false;
        bool min = IsIconic(h);
        RECT r = new RECT();
        if (!min)
        {
            if (DwmGetWindowAttribute(h, 9, out r, 16) != 0) return false;
            if (r.R - r.L < 160 || r.B - r.T < 100) return false;
        }
        info = new Info();
        info.H = h;
        info.Pid = pid;
        info.R = r;
        info.Min = min;
        info.Zoomed = IsZoomed(h);
        return true;
    }

    }

    // ikon: pakai cache; kalau belum ada, dibuat di thread latar (animasi tidak menunggu)
    static string IconQuick(string id)
    {
        if (string.IsNullOrEmpty(id)) return "";
        string v;
        lock (iconCache)
        {
            if (iconCache.TryGetValue(id, out v)) return v;
            if (iconBusy.Contains(id)) return "";
            iconBusy.Add(id);
        }
        string key = id;
        Thread t = new Thread(delegate ()
        {
            string r = "";
            try
            {
                byte[] png = IconUtil.Png(key, 96);
                if (png != null) r = Convert.ToBase64String(png);
            }
            catch (Exception) { }
            lock (iconCache)
            {
                if (iconCache.Count > 150) iconCache.Clear();
                if (r.Length > 0) iconCache[key] = r;
                iconBusy.Remove(key);
            }
        });
        t.IsBackground = true;
        t.SetApartmentState(ApartmentState.STA);
        t.Start();
        return "";
    }

    static string NameOf(string exe)
    {
        if (string.IsNullOrEmpty(exe)) return "";
        string v;
        if (nameCache.TryGetValue(exe, out v)) return v;
        v = "";
        try
        {
            FileVersionInfo vi = FileVersionInfo.GetVersionInfo(exe);
            if (!string.IsNullOrEmpty(vi.FileDescription)) v = vi.FileDescription.Trim();
            if (v.Length == 0 && !string.IsNullOrEmpty(vi.ProductName)) v = vi.ProductName.Trim();
        }
        catch (Exception) { }
        if (nameCache.Count > 300) nameCache.Clear();
        nameCache[exe] = v;
        return v;
    }

    static void Fill(Info i)
    {
        uint rp = RealPid(i.H, i.Pid);
        string[] pi = ProcInfo(rp);
        i.Exe = pi[0];
        i.IconId = pi[1].Length > 0 ? pi[1] : pi[0];
        i.Icon = IconQuick(i.IconId);
        i.Name = NameOf(pi[0]);
    }

    static void Poll(uint own, bool armed, bool first)
    {
        {
            List<string> events = new List<string>();
            Dictionary<IntPtr, Info> now = new Dictionary<IntPtr, Info>();
            HashSet<IntPtr> away = new HashSet<IntPtr>();
            try
            {
                EnumWindows(delegate (IntPtr h, IntPtr l)
                {
                    Info c;
                    int hid;
                    if (Candidate(h, own, out c, out hid)) now[h] = c;
                    else if (hid == 2) away.Add(h);
                    return true;
                }, IntPtr.Zero);

                foreach (KeyValuePair<IntPtr, Info> kv in now)
                {
                    Info c = kv.Value;
                    Info k;
                    if (known.TryGetValue(kv.Key, out k))
                    {
                        k.Away = false;
bool wasMin = k.Min;
                        bool recent = unchecked(Environment.TickCount - k.Hk) < 500;
                        k.Zoomed = c.Zoomed;
                        k.Min = c.Min;
                        if (!c.Min) { k.Prev = k.R; k.R = c.R; }
                        if (k.Announced)
                        {
if (recent) { }
                            else if (!wasMin && c.Min) events.Add(Ev("minimize", k));
                            else if (wasMin && !c.Min) events.Add(Ev("restore", k));
                        }
                        else if (!c.Min)
                        {
                            bool same = k.Prev.L == k.R.L && k.Prev.T == k.R.T && k.Prev.R == k.R.R && k.Prev.B == k.R.B;
                            k.Stable = same ? k.Stable + 1 : 0;
if (k.Stable >= 1)
                            {
                                k.Announced = true;
                                if (armed) { Fill(k); events.Add(Ev("open", k)); }
                                else Fill(k);
                            }
                        }
                    }
                    else
                    {
                        c.Prev = new RECT();
                        // jendela yang sudah ada saat mulai: dianggap sudah dibuka, tanpa animasi
                        if (first || !armed)
                        {
                            c.Announced = true;
                            Fill(c);
                        }
                        known[kv.Key] = c;
                    }
                }

                List<IntPtr> gone = new List<IntPtr>();
                foreach (KeyValuePair<IntPtr, Info> kv in known)
                {
                    if (now.ContainsKey(kv.Key)) continue;
                    if (away.Contains(kv.Key)) { kv.Value.Away = true; continue; }
                    Info k = kv.Value;
                    if (k.Announced && !k.Min && !k.Away && armed) events.Add(Ev("close", k));
                    gone.Add(kv.Key);
                }
                foreach (IntPtr g in gone) known.Remove(g);
            }
            catch (Exception) { }
            // lonjakan peristiwa (mis. bangun dari tidur, ganti monitor) bukan buka/tutup sungguhan
            if (events.Count > 3) events.Clear();
            foreach (string e in events) Program.Emit(e);
        }
    }

    delegate void HookProc(IntPtr hook, uint ev, IntPtr hwnd, int idObj, int idChild, uint thr, uint time);
    [StructLayout(LayoutKind.Sequential)] struct PT { public int X, Y; }
    [StructLayout(LayoutKind.Sequential)] struct MSG { public IntPtr H; public uint M; public UIntPtr W; public IntPtr L; public uint T; public PT P; }
    [DllImport("user32.dll")] static extern IntPtr SetWinEventHook(uint min, uint max, IntPtr mod, HookProc p, uint pid, uint tid, uint flags);
    [DllImport("user32.dll")] static extern bool PeekMessage(out MSG m, IntPtr h, uint a, uint b, uint remove);
    [DllImport("user32.dll")] static extern bool TranslateMessage(ref MSG m);
    [DllImport("user32.dll")] static extern IntPtr DispatchMessage(ref MSG m);
    [DllImport("user32.dll")] static extern uint MsgWaitForMultipleObjects(uint n, IntPtr[] h, bool all, uint ms, uint mask);

    static HookProc hookProc;
    static readonly List<IntPtr> pendShow = new List<IntPtr>();
    static readonly List<IntPtr> pendHide = new List<IntPtr>();
    static readonly List<IntPtr> pendMin = new List<IntPtr>();
    static readonly List<IntPtr> pendRes = new List<IntPtr>();

    static void OnHook(IntPtr hook, uint ev, IntPtr hwnd, int idObj, int idChild, uint thr, uint time)
    {
        if (idObj != 0 || idChild != 0 || hwnd == IntPtr.Zero) return;
if (ev == 0x16) pendMin.Add(hwnd);
        else if (ev == 0x17) pendRes.Add(hwnd);
        else if (ev == 0x8002) pendShow.Add(hwnd);
        else if (ev == 0x8003) pendHide.Add(hwnd);
    }

    // jalur cepat: dipanggil begitu jendela muncul (tanpa menunggu polling)
    static void OnShow(IntPtr h, uint own, bool armed)
    {
        if (known.ContainsKey(h)) return;
        Info c;
        int hid;
        if (!Candidate(h, own, out c, out hid) || c.Min) return;
        c.Prev = c.R;
        c.Stable = 9;
        c.Announced = true;
        Fill(c);
        known[h] = c;
        if (armed) Program.Emit(Ev("open", c));
    }

    // jalur cepat minimize / pulihkan: langsung dari sistem, tanpa menunggu polling
    static void OnMin(IntPtr h, bool armed)
    {
        Info k;
        if (!known.TryGetValue(h, out k)) return;
        if (!k.Announced || k.Min) return;
        k.Min = true;
        k.Hk = Environment.TickCount;
        if (armed) Program.Emit(Ev("minimize", k));
    }

    static void OnRes(IntPtr h, uint own, bool armed)
    {
        Info k;
        if (!known.TryGetValue(h, out k)) return;
        if (!k.Announced || !k.Min) return;
        Info c;
        int hid;
        if (Candidate(h, own, out c, out hid) && !c.Min) { k.R = c.R; k.Zoomed = c.Zoomed; }
        k.Min = false;
        k.Hk = Environment.TickCount;
        if (armed) Program.Emit(Ev("restore", k));
    }

    static void OnHide(IntPtr h, bool armed)
    {
        Info k;
        if (!known.TryGetValue(h, out k)) return;
        if (IsWindowVisible(h) || IsIconic(h)) return;
        if (CloakKind(h) == 2) return;
        known.Remove(h);
        if (k.Announced && !k.Min && !k.Away && armed) Program.Emit(Ev("close", k));
    }

    public static void Run(uint own)
    {
        DateTime start = DateTime.UtcNow;
        DateTime lastPoll = DateTime.MinValue;
        bool first = true;
        hookProc = new HookProc(OnHook);
        IntPtr hk = IntPtr.Zero;
        try { hk = SetWinEventHook(0x8002, 0x8003, IntPtr.Zero, hookProc, 0, 0, 2); } catch (Exception) { }
        IntPtr hk2 = IntPtr.Zero;
        try { hk2 = SetWinEventHook(0x0016, 0x0017, IntPtr.Zero, hookProc, 0, 0, 2); } catch (Exception) { }
        while (true)
        {
            MSG m;
            while (PeekMessage(out m, IntPtr.Zero, 0, 0, 1)) { TranslateMessage(ref m); DispatchMessage(ref m); }
            bool armed = (DateTime.UtcNow - start).TotalMilliseconds > 1500;
            try
            {
                if (pendShow.Count > 6 || pendHide.Count > 6) { pendShow.Clear(); pendHide.Clear(); }
                List<IntPtr> mn = new List<IntPtr>(pendMin); pendMin.Clear();
                List<IntPtr> rs = new List<IntPtr>(pendRes); pendRes.Clear();
                foreach (IntPtr h in mn) OnMin(h, armed);
                foreach (IntPtr h in rs) OnRes(h, own, armed);
                List<IntPtr> sh = new List<IntPtr>(pendShow); pendShow.Clear();
                List<IntPtr> hi = new List<IntPtr>(pendHide); pendHide.Clear();
                foreach (IntPtr h in hi) OnHide(h, armed);
                foreach (IntPtr h in sh) OnShow(h, own, armed);
            }
            catch (Exception) { }
            double since = (DateTime.UtcNow - lastPoll).TotalMilliseconds;
            if (hk == IntPtr.Zero ? since >= 45 : since >= 60)
            {
                lastPoll = DateTime.UtcNow;
                Poll(own, armed, first);
                first = false;
            }
            MsgWaitForMultipleObjects(0, null, false, 8, 0x04FF);
        }
    }
}

static class MbBar
{
    [StructLayout(LayoutKind.Sequential)] struct RECT { public int L, T, R, B; }
    [StructLayout(LayoutKind.Sequential)] struct APPBARDATA { public int cbSize; public IntPtr hWnd; public uint uCallbackMessage; public uint uEdge; public RECT rc; public IntPtr lParam; }
    [DllImport("shell32.dll")] static extern UIntPtr SHAppBarMessage(uint msg, ref APPBARDATA d);

    {
        if (cur == IntPtr.Zero) return;
        APPBARDATA d = new APPBARDATA();

static class Program
{
    static readonly object O = new object();
    public static volatile bool TopOn = false;
    public static volatile bool PingOn = false;
    public static volatile bool BtnOn = false;

    public static void Emit(string json)
    {
        lock (O)
        {
            Console.Out.WriteLine(json);
            Console.Out.Flush();
        }
    }

    public static string J(string s)
    {
        if (s == null) return "\"\"";
        StringBuilder sb = new StringBuilder();
        sb.Append('"');
        foreach (char c in s)
        {
            if (c == '"') sb.Append("\\\"");
            else if (c == '\\') sb.Append("\\\\");
            else if (c == '\n') sb.Append("\\n");
            else if (c == '\r') sb.Append("\\r");
            else if (c == '\t') sb.Append("\\t");
            else if (c < 32 || c > 126) sb.Append("\\u").Append(((int)c).ToString("x4"));
            else sb.Append(c);
        }
        sb.Append('"');
        return sb.ToString();
    }

    static void Spawn(ThreadStart f)
    {
        Thread t = new Thread(f);
        t.IsBackground = true;
        t.Start();
    }

    static int Main(string[] a)
    {
        try
        {
            Spawn(delegate () { Guard(Sys.NetRes); });
            Spawn(delegate () { Guard(Sys.Gpu); });
            Spawn(delegate () { Guard(Sys.Top); });
            Spawn(delegate () { Guard(Sys.Devices); });
            Spawn(delegate () { Guard(Sys.Audio); });
            Spawn(delegate () { Guard(Sys.Progress); });
            Spawn(delegate () { Guard(Sys.Game); });
            Spawn(delegate () { Guard(Sys.PingLoop); });
            Spawn(delegate () { Guard(Sys.Mouse); });
            Spawn(delegate () { Guard(Sys.Clip); });
            Spawn(delegate () { Guard(Sys.Volume); });
            Emit("{\"t\":\"ready\"}");
            string line;
            while ((line = Console.In.ReadLine()) != null)
            {
                try { Command(line.Trim()); } catch (Exception) { }
            }
        }
        catch (Exception e)
        {
            Emit("{\"t\":\"err\",\"m\":" + J(e.GetType().Name + ": " + e.Message) + "}");
        }
        try { MbBar.Off(); } catch (Exception) { }
        try { CursorFx.Restore(); } catch (Exception) { }
        Environment.Exit(0);
        return 0;
    }

    static void Guard(ThreadStart f)
    {
        try { f(); }
        catch (Exception e) { Emit("{\"t\":\"err\",\"m\":" + J(f.Method.Name + ": " + e.Message) + "}"); }
    }

    static void Command(string l)
    {
        if (l.Length == 0) return;
        string[] p = l.Split(' ');
        string c = p[0];
        if (c == "paste") Keys.Paste();
        else if (c == "media" && p.Length > 1) Keys.Media(p[1]);
        else if (c == "top") TopOn = p.Length > 1 && p[1] == "1";
        else if (c == "ping") PingOn = p.Length > 1 && p[1] == "1";
        else if (c == "btn") BtnOn = p.Length > 1 && p[1] == "1";
        else if (c == "games") Sys.Extra = Sys.Names(p.Length > 1 ? p[1] : "");
        else if (c == "nogames") Sys.NoGame = Sys.Names(p.Length > 1 ? p[1] : "");
        else if (c == "wins") Spawn(delegate () { Guard(Sys.Wins); });
        else if (c == "vol" && p.Length > 1) { int vv; if (int.TryParse(p[1], out vv)) Sys.SetVol(vv); }
        else if (c == "mute") Sys.ToggleMute();
        else if (c == "theme") { try { MbTheme.Broadcast(); } catch (Exception) { } }
        else if (c == "bar")
        {
            if (p.Length > 5)
            {
                long hw; int bx, by, bw, bh;
                if (long.TryParse(p[1], out hw) && int.TryParse(p[2], out bx) && int.TryParse(p[3], out by) && int.TryParse(p[4], out bw) && int.TryParse(p[5], out bh)) MbBar.Set(hw, bx, by, bw, bh);
            }
            else MbBar.Off();
        }
        else if (c == "cursor")
        {
            if (p.Length > 1 && p[1] == "off") { CursorFx.Restore(); return; }
            int sz = 32;
            if (p.Length > 3) int.TryParse(p[3], out sz);
            CursorFx.Apply(p.Length > 1 ? p[1] : "arrow", p.Length > 2 ? p[2] : "ffffff", sz);
        }
    }
}

    public static bool CoversMonitor(IntPtr h)
    {
        RECT r;
        if (!GetWindowRect(h, out r)) return false;
        IntPtr mon = MonitorFromWindow(h, 2);
        if (mon == IntPtr.Zero) return false;
        MONITORINFO mi = new MONITORINFO();
        mi.cbSize = Marshal.SizeOf(typeof(MONITORINFO));
        if (!GetMonitorInfo(mon, ref mi)) return false;
        return r.L <= mi.rcMonitor.L + 1 && r.T <= mi.rcMonitor.T + 1 && r.R >= mi.rcMonitor.R - 1 && r.B >= mi.rcMonitor.B - 1;
    }

static class Sys
{
    [StructLayout(LayoutKind.Sequential)]
    struct FILETIME { public uint Lo; public uint Hi; }
    [DllImport("kernel32.dll")] static extern bool GetSystemTimes(out FILETIME idle, out FILETIME kernel, out FILETIME user);

    [StructLayout(LayoutKind.Sequential)]
    class MEMSTAT
    {
        public uint dwLength = (uint)Marshal.SizeOf(typeof(MEMSTAT));
        public uint dwMemoryLoad;
        public ulong ullTotalPhys;
        public ulong ullAvailPhys;
        public ulong ullTotalPageFile;
        public ulong ullAvailPageFile;
        public ulong ullTotalVirtual;
        public ulong ullAvailVirtual;
        public ulong ullAvailExtendedVirtual;
    }
    [DllImport("kernel32.dll")] static extern bool GlobalMemoryStatusEx([In, Out] MEMSTAT m);

    static ulong Ft(FILETIME f) { return ((ulong)f.Hi << 32) | f.Lo; }

    public static volatile int GpuNow = -1;

    // ----- jaringan, CPU, RAM (tiap 1 detik)
    public static void NetRes()
    {
        long lastRx = -1, lastTx = -1;
        ulong pi = 0, pk = 0, pu = 0;
        DateTime lastT = DateTime.UtcNow;
        while (true)
        {
            Thread.Sleep(1000);
            DateTime now = DateTime.UtcNow;
            double dt = Math.Max(0.2, (now - lastT).TotalSeconds);
            lastT = now;
            long rx = 0, tx = 0;
            try
            {
                foreach (NetworkInterface ni in NetworkInterface.GetAllNetworkInterfaces())
                {
                    if (ni.OperationalStatus != OperationalStatus.Up) continue;
                    if (ni.NetworkInterfaceType == NetworkInterfaceType.Loopback || ni.NetworkInterfaceType == NetworkInterfaceType.Tunnel) continue;
                    IPv4InterfaceStatistics st = ni.GetIPv4Statistics();
                    rx += st.BytesReceived;
                    tx += st.BytesSent;
                }
            }
            catch (Exception) { }
            long d = 0, u = 0;
            if (lastRx >= 0 && rx >= lastRx && tx >= lastTx)
            {
                d = (long)((rx - lastRx) / dt);
                u = (long)((tx - lastTx) / dt);
            }
            lastRx = rx;
            lastTx = tx;
            Program.Emit("{\"t\":\"net\",\"d\":" + d + ",\"u\":" + u + "}");

            int cpu = 0;
            try
            {
                FILETIME fi, fk, fu;
                if (GetSystemTimes(out fi, out fk, out fu))
                {
                    ulong i = Ft(fi), k = Ft(fk), us = Ft(fu);
                    if (pk != 0)
                    {
                        ulong dI = i - pi, dK = k - pk, dU = us - pu;
                        ulong tot = dK + dU;
                        if (tot > 0) cpu = (int)Math.Max(0, Math.Min(100, 100.0 * (tot - dI) / tot));
                    }
                    pi = i; pk = k; pu = us;
                }
            }
            catch (Exception) { }
            int ram = 0;
            try
            {
                MEMSTAT m = new MEMSTAT();

    public static void Gpu()
    {

            {

                {
                    foreach (PerformanceCounter pc in cs) { try { pc.Dispose(); } catch (Exception) { } }
                    cs.Clear();
                    PerformanceCounterCategory cat = new PerformanceCounterCategory("GPU Engine");
                    foreach (string inst in cat.GetInstanceNames())
                    {
                        if (inst.IndexOf("engtype_3D", StringComparison.OrdinalIgnoreCase) < 0) continue;
                        PerformanceCounter pc2 = new PerformanceCounter("GPU Engine", "Utilization Percentage", inst, true);
                        try { pc2.NextValue(); cs.Add(pc2); } catch (Exception) { pc2.Dispose(); }
                        if (cs.Count > 120) break;

                    built = DateTime.UtcNow;

                }
                double sum = 0;
                for (int i = 0; i < cs.Count; i++)
                {
                    try { sum += cs[i].NextValue(); } catch (Exception) { }
                }
                GpuNow = cs.Count == 0 ? -1 : (int)Math.Max(0, Math.Min(100, sum));
            }
            catch (Exception) { GpuNow = -1; failed = true; }
        }
    }

    public static void Top()

        {
            Thread.Sleep(2000);
            if (!Program.TopOn) continue;
            try
            {
                List<KeyValuePair<long, string>> l = new List<KeyValuePair<long, string>>();
                int me = Process.GetCurrentProcess().Id;
                foreach (Process p in Process.GetProcesses())
                {
                    try
                    {
                        if (p.Id <= 4 || p.Id == me) continue;
                        long ws = p.WorkingSet64;
                        string n = p.ProcessName;
                        l.Add(new KeyValuePair<long, string>(ws, p.Id + "|" + n));
                    }
                    catch (Exception) { }
                    finally { p.Dispose(); }
                }
                l.Sort(delegate (KeyValuePair<long, string> x, KeyValuePair<long, string> y) { return y.Key.CompareTo(x.Key); });
                StringBuilder sb = new StringBuilder();
                sb.Append("{\"t\":\"top\",\"l\":[");
                int c = 0;
                foreach (KeyValuePair<long, string> kv in l)
                {
                    if (c >= 4) break;
                    string[] pr = kv.Value.Split('|');
                    if (c > 0) sb.Append(',');
                    sb.Append("{\"pid\":").Append(pr[0]).Append(",\"n\":").Append(Program.J(pr[1])).Append(",\"mb\":").Append(kv.Key / (1024 * 1024)).Append('}');
                    c++;
                }
                sb.Append("]}");
                Program.Emit(sb.ToString());
            }
            catch (Exception) { }
        }
    }

    static List<string> InUse(string cap)
    {
        List<string> res = new List<string>();
        string b = "Software\\Microsoft\\Windows\\CurrentVersion\\CapabilityAccessManager\\ConsentStore\\" + cap;
        string[] roots = new string[] { b, b + "\\NonPackaged" };
        foreach (string root in roots)
        {
            try
            {
                using (RegistryKey rk = Registry.CurrentUser.OpenSubKey(root))
                {
                    if (rk == null) continue;
                    foreach (string sn in rk.GetSubKeyNames())
                    {
                        if (sn == "NonPackaged") continue;
                        try
                        {
                            using (RegistryKey k = rk.OpenSubKey(sn))
                            {
                                if (k == null) continue;
                                object s = k.GetValue("LastUsedTimeStart");
                                object e = k.GetValue("LastUsedTimeStop");
                                if (s == null || e == null) continue;
                                long ls = Convert.ToInt64(s), le = Convert.ToInt64(e);
                                if (ls > 0 && le == 0) res.Add(Nice(sn));
                            }
                        }
                        catch (Exception) { }
                    }
                }
            }
            catch (Exception) { }
        }
        return res;
    }

    static string Nice(string sn)
    {
        string n = sn;
        int i = n.LastIndexOf('#');
        if (i >= 0) n = n.Substring(i + 1);
        else
        {
            int u = n.IndexOf('_');
            if (u > 0) n = n.Substring(0, u);
            int d = n.LastIndexOf('.');
            if (d >= 0 && d < n.Length - 1) n = n.Substring(d + 1);
        }
        if (n.EndsWith(".exe", StringComparison.OrdinalIgnoreCase)) n = n.Substring(0, n.Length - 4);
        return n;
    }

    static string[] FindMedia()
    {
        List<Wnd.Item> ws = Wnd.Visible();
        string[] spot = null, web = null, vlc = null;
        foreach (Wnd.Item w in ws)
        {
            string t = w.Title;
            if (w.Exe == "spotify")
            {
                if (t == "Spotify" || t.StartsWith("Spotify Free") || t.StartsWith("Spotify Premium") || t.Length == 0) { if (spot == null) spot = new string[] { "Spotify", "", "" }; continue; }
                int i = t.IndexOf(" - ", StringComparison.Ordinal);
                spot = i > 0 ? new string[] { "Spotify", t.Substring(i + 3), t.Substring(0, i) } : new string[] { "Spotify", t, "" };
            }
            else if (Array.IndexOf(BROWSERS, w.Exe) >= 0 && web == null)
            {
                string core = Trim(t, BSUF);
                if (core == null) continue;
                string site = Trim(core, SITES);
                if (site == null) continue;
                string app = core.EndsWith("YouTube Music") ? "YouTube Music" : (core.EndsWith("SoundCloud") ? "SoundCloud" : (core.EndsWith("Spotify") ? "Spotify" : "YouTube"));
                web = new string[] { app, site, "" };
            }
            else if (w.Exe == "vlc" && vlc == null)
            {
                string core = Trim(t, new string[] { " - VLC media player" });
                if (core != null) vlc = new string[] { "VLC", core, "" };
            }
        }
        if (spot != null && spot[1].Length > 0) return spot;
        if (web != null) return web;
        if (vlc != null) return vlc;
        return spot;
    }

                    ne = Wnd.ExeName(pid);
                    nf = IsGame(ne, FullPath(pid));
                }
            }
            catch (Exception) { }
            string ge = nf ? ne : "";
            if (nf != full || ge != exe || ne != curLast)
            {
                full = nf; exe = ge; curLast = ne;
                Program.Emit("{\"t\":\"fg\",\"full\":" + (full ? 1 : 0) + ",\"exe\":" + Program.J(exe) + ",\"cur\":" + Program.J(ne) + "}");
                lastF = 0;
            }
            if (full)
            {
                try
                {
                    ulong f = ReadFrame(buf, ref good);
                    DateTime now = DateTime.UtcNow;
                    double dt = (now - lastT).TotalSeconds;
                    lastT = now;
                    int fps = -1;
                    if (f != 0 && lastF != 0 && f >= lastF && dt > 0.2) fps = (int)Math.Round((f - lastF) / dt);
                    if (f != 0) lastF = f;
                    if (fps >= 0 && fps < 1000) Program.Emit("{\"t\":\"fps\",\"v\":" + fps + "}");
                }
                catch (Exception) { }
            }
        }
    }

    static readonly Dictionary<uint, bool> own = new Dictionary<uint, bool>();
    static string root = null;
    // jendela milik aplikasi ini sendiri (Dock, island, dll) tidak perlu dipaksa: kursornya sudah kursor kita
    static bool IsOwn(uint pid)
    {
        bool r;
        if (own.TryGetValue(pid, out r)) return r;
        r = false;
        try
        {
            if (root == null)
            {
                string b = AppDomain.CurrentDomain.BaseDirectory.TrimEnd('\\', '/');
                root = System.IO.Path.GetDirectoryName(b);
                root = (root == null ? "" : root.TrimEnd('\\', '/') + "\\");
            }
            string f = Process.GetProcessById((int)pid).MainModule.FileName;
            if (root.Length > 3 && f.StartsWith(root, StringComparison.OrdinalIgnoreCase)) r = true;
        }
        catch (Exception) { }
        if (own.Count > 300) own.Clear();
        own[pid] = r;
        return r;
    }

    // semua kursor sistem: panah, I-beam, tunggu, silang, panah atas, ubah ukuran (4 arah + semua), dilarang, tangan, sibuk, bantuan, pin, orang
    static readonly uint[] IDS = new uint[] { 32512, 32513, 32514, 32515, 32516, 32642, 32643, 32644, 32645, 32646, 32648, 32649, 32650, 32651, 32671, 32672 };   // SEMUA kursor sistem, dipasang sekali (tanpa pengawas/pemasangan ulang) supaya tidak berganti-ganti

    static Color Parse(string hex)
    {
        try
        {
            hex = hex.TrimStart('#');
            if (hex.Length == 6) return Color.FromArgb(Convert.ToInt32(hex.Substring(0, 2), 16), Convert.ToInt32(hex.Substring(2, 2), 16), Convert.ToInt32(hex.Substring(4, 2), 16));
        }
        catch (Exception) { }
        return Color.White;
    }

    static Bitmap Draw(string style, Color c, int s, out int hx, out int hy)
    {
        Bitmap bmp = new Bitmap(s, s, System.Drawing.Imaging.PixelFormat.Format32bppArgb);
        using (Graphics g = Graphics.FromImage(bmp))
        {
            g.SmoothingMode = SmoothingMode.AntiAlias;
            g.Clear(Color.Transparent);
            hx = 2; hy = 2;
            float m = 2f, u = s - 5f;
            using (Pen edge = new Pen(Color.FromArgb(230, 20, 20, 24), 2.2f))
            using (Brush fill = new SolidBrush(c))
            using (Pen line = new Pen(c, 2.6f))
            {
                edge.LineJoin = LineJoin.Round;
                if (style == "ring")
                {
                    hx = s / 2; hy = s / 2;
                    float d = s * 0.62f, o = (s - d) / 2f;
                    g.DrawEllipse(edge, o - 1f, o - 1f, d + 2f, d + 2f);
                    g.DrawEllipse(line, o, o, d, d);
                    g.FillEllipse(fill, s / 2f - 2f, s / 2f - 2f, 4f, 4f);
                }
                else if (style == "dot")
                {
                    hx = s / 2; hy = s / 2;
                    float d = s * 0.34f, o = (s - d) / 2f;
                    g.FillEllipse(new SolidBrush(Color.FromArgb(235, 20, 20, 24)), o - 1.5f, o - 1.5f, d + 3f, d + 3f);
                    g.FillEllipse(fill, o, o, d, d);
                }
                else if (style == "cross")
                {
                    hx = s / 2; hy = s / 2;
                    float h = s / 2f, gap = s * 0.12f, e = s * 0.42f;
                    g.DrawLine(edge, h - e, h, h - gap, h); g.DrawLine(edge, h + gap, h, h + e, h);
                    g.DrawLine(edge, h, h - e, h, h - gap); g.DrawLine(edge, h, h + gap, h, h + e);
                    g.DrawLine(line, h - e, h, h - gap, h); g.DrawLine(line, h + gap, h, h + e, h);
                    g.DrawLine(line, h, h - e, h, h - gap); g.DrawLine(line, h, h + gap, h, h + e);
                }
                else
                {
                    PointF[] pts = new PointF[] {
                        new PointF(m, m), new PointF(m, m + u * 0.86f), new PointF(m + u * 0.22f, m + u * 0.68f),
                        new PointF(m + u * 0.36f, m + u), new PointF(m + u * 0.5f, m + u * 0.93f), new PointF(m + u * 0.36f, m + u * 0.63f),
                        new PointF(m + u * 0.64f, m + u * 0.63f) };
                    g.DrawPolygon(edge, pts);
                    g.FillPolygon(fill, pts);
                }
            }
        }
        return bmp;
    }

    static IntPtr MakeCursor(Bitmap bmp, int hx, int hy)
    {
        IntPtr hi = bmp.GetHicon();
        ICONINFO ii;
        if (!GetIconInfo(hi, out ii)) { DestroyIcon(hi); return IntPtr.Zero; }
        ii.fIcon = false;
        ii.xHotspot = hx;
        ii.yHotspot = hy;
        IntPtr cur = CreateIconIndirect(ref ii);
        if (ii.hbmMask != IntPtr.Zero) DeleteObject(ii.hbmMask);
        if (ii.hbmColor != IntPtr.Zero) DeleteObject(ii.hbmColor);
        DestroyIcon(hi);
        return cur;
    }

    // dipanggil di dalam lock(L): ganti SEMUA kursor sistem dengan gaya pilihan + siapkan kursor cadangan
    static void Build()
    {
        Color c = Parse(hexc);
        int hx, hy;
        using (Bitmap bmp = Draw(st, c, sz, out hx, out hy))
        {
            for (int i = 0; i < IDS.Length; i++)
            {
                IntPtr cur = MakeCursor(bmp, hx, hy);
                if (cur == IntPtr.Zero) continue;
                if (SetSystemCursor(cur, IDS[i])) changed = true;
                else DestroyIcon(cur);
            }
            IntPtr old = force;
            force = MakeCursor(bmp, hx, hy);
            if (old != IntPtr.Zero) DestroyIcon(old);
        }
        for (int i = 0; i < IDS.Length; i++) sys[i] = LoadCursor(IntPtr.Zero, (IntPtr)IDS[i]);
    }

    static bool Ours(IntPtr h)
    {
        if (h == force) return true;
        for (int i = 0; i < sys.Length; i++) if (sys[i] == h) return true;
        return false;
    }

    // Pengawas: kalau Windows/aplikasi/game mengembalikan kursor ke bawaan (reset), pasang lagi. Tetap dipasang ulang berkala sebagai jaring pengaman.
    static void Watch()
    {
        int tick = 0;
        while (true)
        {
            Thread.Sleep(500);
            try
            {
                lock (L)
                {
                    if (!on) continue;
                    tick++;
                    if (tick < 10) continue;
                    bool bad = false;
                    for (int i = 0; i < IDS.Length && !bad; i++)
                        if (sys[i] != IntPtr.Zero && LoadCursor(IntPtr.Zero, (IntPtr)IDS[i]) != sys[i]) bad = true;
                    if (bad) { tick = 0; Build(); }
                }
            }
            catch (Exception) { }
        }
    }

    // Pemaksa: game seperti Roblox memasang kursor gambarnya sendiri (jari, panah putih, dll) yang tidak ikut terganti.
    // Setiap kali kursor yang tampil bukan milik kita, langsung ditimpa dengan kursor kita. Tidak menyuntik apa pun ke game.
    static void Force()
    {
        uint me = GetCurrentThreadId();
        while (true)
        {
            Thread.Sleep(3);
            try
            {
                if (!on) { Thread.Sleep(100); continue; }
                CURSORINFO ci = new CURSORINFO();
                ci.cbSize = Marshal.SizeOf(typeof(CURSORINFO));
                if (!GetCursorInfo(ref ci)) continue;
                if ((ci.flags & 1) == 0 || ci.hCursor == IntPtr.Zero) continue;    // kursor sengaja disembunyikan (mouse-lock / first person): biarkan
                if ((GetAsyncKeyState(0x01) & 0x8000) != 0 || (GetAsyncKeyState(0x02) & 0x8000) != 0) continue;   // jangan ganggu saat tombol mouse ditekan (klik/seret)
                lock (L)
                {
                    if (!on || force == IntPtr.Zero || Ours(ci.hCursor)) continue;
                    IntPtr w = WindowFromPoint(ci.pt);
                    if (w == IntPtr.Zero) continue;
                    uint pid;
                    uint tid = GetWindowThreadProcessId(w, out pid);
                    if (tid == 0 || tid == me || IsOwn(pid)) continue;
                    bool att = AttachThreadInput(me, tid, true);
                    SetCursor(force);
                    if (att) AttachThreadInput(me, tid, false);
                }
            }
            catch (Exception) { }
        }
    }

    public static void Apply(string style, string hex, int size)
    {
        if (size < 20) size = 20;
        if (size > 64) size = 64;
        lock (L)
        {
            st = (style == "arrow" || style == "dot" || style == "cross") ? style : "ring";
            hexc = hex; sz = size;
            on = true;
            Build();
        }
    }

    public static void Restore()
    {
        lock (L)
        {
            on = false;
            if (!changed) return;
            changed = false;
            SystemParametersInfo(0x0057, 0, IntPtr.Zero, 0);
        }
    }

    function springTo(w, h, opening) {
      S.tw = w; S.th = h;
      if (opening) { S.kw = 230; S.dw = 16; S.kh = 270; S.dh = 18; } else { S.kw = 420; S.dw = 29; S.kh = 420; S.dh = 29; }
      if (!S.run) { S.run = true; S.last = 0; requestAnimationFrame(frame); }
    }

    function frame(t) {
      if (!S.run) return;
      var dt = S.last ? Math.min((t - S.last) / 1000, 1 / 30) : 1 / 60; S.last = t;
      for (var i = 0; i < 4; i++) {
        var d = dt / 4;
        S.vw += (-S.kw * (S.w - S.tw) - S.dw * S.vw) * d; S.w += S.vw * d;
        S.vh += (-S.kh * (S.h - S.th) - S.dh * S.vh) * d; S.h += S.vh * d;
      }
      var done = Math.abs(S.w - S.tw) < 0.4 && Math.abs(S.h - S.th) < 0.4 && Math.abs(S.vw) < 3 && Math.abs(S.vh) < 3;
      if (done) {
        S.w = S.tw; S.h = S.th; S.vw = 0; S.vh = 0; S.run = false;
        if (openingFx) { openingFx = false; ct.style.opacity = 1; ct.style.scale = 1; ct.style.filter = 'none'; }
      }
      applySize(S.w, S.h);
      progress();
      if (done) { if (!open && !keepVisible()) { isl.classList.remove('on'); if (o.onIdle) o.onIdle(); } return; }
      requestAnimationFrame(frame);
    }

    function tint(src) {
      if (!src || !C.glow) { glowRgb = null; style(); return; }
      var im = new Image();
      im.onload = function () {
        try {
          var s = 20, cv = document.createElement('canvas'); cv.width = cv.height = s;
          var cx = cv.getContext('2d'); cx.drawImage(im, 0, 0, s, s);
          var d = cx.getImageData(0, 0, s, s).data, r = 0, g2 = 0, b = 0, n = 0;
          for (var i = 0; i < d.length; i += 4) {
            if (d[i + 3] < 200) continue;
            var mx = Math.max(d[i], d[i + 1], d[i + 2]), mn = Math.min(d[i], d[i + 1], d[i + 2]);
            if (mx - mn < 40) continue;
            r += d[i]; g2 += d[i + 1]; b += d[i + 2]; n++;
          }
          glowRgb = n > 3 ? [Math.round(r / n), Math.round(g2 / n), Math.round(b / n)] : null;
        } catch (x) { glowRgb = null; }
