}

// Cadangan kalau Get-StartApps gagal/lambat: baca pintasan (.lnk) di folder Start Menu.
function scanStartMenu(list, seen) {
  const roots = [path.join(process.env.ProgramData || "C:\\ProgramData", "Microsoft", "Windows", "Start Menu", "Programs"),
                 path.join(process.env.APPDATA || "", "Microsoft", "Windows", "Start Menu", "Programs")];
  const walk = (d, depth) => {
    if (depth > 4 || !d) return;
    let es = [];
    try { es = fs.readdirSync(d, { withFileTypes: true }); } catch (_) { return; }
    es.forEach(e => {
      const f = path.join(d, e.name);
      if (e.isDirectory()) { walk(f, depth + 1); return; }
      if (!/\.lnk$/i.test(e.name)) return;
      const name = e.name.replace(/\.lnk$/i, "");
      if (/uninstall|hapus|readme|license/i.test(name) || seen.has(f)) return;
      seen.add(f); list.push({ n: name, p: f }); appNames.set(f, name);
    });
  };
  roots.forEach(r => walk(r, 0));
}
function refreshApps(tries) {
  tries = tries | 0;
  const ps = "[Console]::OutputEncoding=[Text.Encoding]::UTF8; Get-StartApps | ForEach-Object { $_.AppID + [char]9 + $_.Name }";
  const done = list => {
    if (!list.length && tries < 2) { setTimeout(() => refreshApps(tries + 1), 4000); return; }   // coba lagi, jangan langsung menyerah
    if (!list.length) scanStartMenu(list, new Set());
    startApps = list; appsReady = true;
  };
  try {
    execFile("powershell.exe", ["-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass", "-Command", ps],
      { windowsHide: true, timeout: 60000, maxBuffer: 16 * 1024 * 1024, encoding: "utf8" }, (err, out) => {
        const list = [], seen = new Set();
        if (out) {
          String(out).split(/\r?\n/).forEach(l => {
            const i = l.indexOf("\t");
            if (i <= 0) return;
            const id = l.slice(0, i).trim(), name = l.slice(i + 1).trim();
            if (!id || !name || seen.has(id)) return;
            seen.add(id); list.push({ n: name, p: id }); appNames.set(id, name);
          });
        }
        done(list);
      });
  } catch (_) { done([]); }
}

function prettyApp(id) {
  if (!id) return "Windows";
  const n = appNames.get(id) || (S.seen && S.seen[id]);
  if (n) return n;
  let s = String(id);
  if (/^[a-z]:[\\/]/i.test(s) || /\.exe$/i.test(s)) s = path.basename(s).replace(/\.exe$/i, "");
  else if (s.indexOf("!") >= 0) { s = s.split("!")[0].split("_")[0].split(".").pop(); }
  else if (s.indexOf(".") >= 0) s = s.split(".").pop();
  s = s.replace(/[_-]+/g, " ").replace(/([a-z])([A-Z])/g, "$1 $2").trim();
  return s ? s.charAt(0).toUpperCase() + s.slice(1) : "Windows";
}

const mediaDir = () => path.join(process.env.WINDIR || "C:\\Windows", "Media");
function soundList() {
  const out = [];
  try {
    fs.readdirSync(mediaDir()).forEach(f => {
      if (!/\.wav$/i.test(f)) return;
      let sz = 0; try { sz = fs.statSync(path.join(mediaDir(), f)).size; } catch (_) {}

    });
  } catch (_) {}
  out.sort((a, b) => a.t.localeCompare(b.t));
  return out;
}

  return l.length ? l[0].u : "";
}

function taskbarOrigin(d, r) {
  const b = d.bounds, w = d.workArea;
  const gb = (b.y + b.height) - (w.y + w.height), gt = w.y - b.y, gl = w.x - b.x, gr = (b.x + b.width) - (w.x + w.width);
  const m = Math.max(gb, gt, gl, gr);
  const mid = b.x + b.width / 2;
  const cx = Math.min(Math.max(r.x + r.width / 2, mid - 260), mid + 260);
  if (m < 8) return { x: cx, y: b.y + b.height - 28 };
  if (m === gb) return { x: cx, y: b.y + b.height - gb / 2 };
  if (m === gt) return { x: cx, y: b.y + gt / 2 };
  if (m === gl) return { x: b.x + gl / 2, y: r.y + r.height / 2 };
  return { x: b.x + b.width - gr / 2, y: r.y + r.height / 2 };
}
function animPlace(o, d, msg) {
  try {
    const b = o.win.getBounds();
    if (b.width !== d.bounds.width || b.height !== d.bounds.height) o.win.setBounds(d.bounds);
    else if (b.x !== d.bounds.x || b.y !== d.bounds.y) o.win.setPosition(d.bounds.x, d.bounds.y, false);
    if (!o.shown) { o.win.setAlwaysOnTop(true, "screen-saver"); o.win.showInactive(); o.shown = true; }
    o.win.webContents.send("fx:play", msg);
  } catch (_) {}
}

function onSys(m) {

  if (m.t === "ready" && !S.enabled) { sysState = { ok: true, err: "" }; sysTries = 0; syncSysFlags(); return; }
  if (!S.enabled) return;
  switch (m.t) {
    case "ready": sysState = { ok: true, err: "" }; sysTries = 0; syncSysFlags(); applyCursor(); break;
    case "err": sysState.err = String(m.m || ""); break;
    case "net": onNet(m); break;
    case "res": onRes(m); break;
    case "top": lastTop = (Array.isArray(m.l) ? m.l : []).filter(x => x && Number.isInteger(x.pid)).slice(0, 4); if (lag && L.mon) setLive("mon", monActivity()); break;
    case "dev": onDev(m); break;
    case "media": onMedia(m); break;
    case "lvl": if (L.media && mediaState.play) toIsl("lvl", m.p | 0); break;
    case "prog": onProg(m); break;
    case "fg": onFg(m); mbSend("app", mbApp()); mbVis(); break;
    case "wins": winsList = (Array.isArray(m.l) ? m.l : []).filter(x => x && typeof x.n === "string").map(x => ({ n: x.n.slice(0, 60), t: String(x.t || "").slice(0, 60) })).slice(0, 40); break;
    case "fps": if (!pm.p || !pm.got) { gameState.fps = m.v | 0; gameState.real = false; refreshGame(); } break;
    case "ping": onPing(m); break;
    case "btn": onBtn(m); break;
    case "clip": clipChanged(true); break;
  }
}

let gmPrio = [];
function gmSetPrio(level) {
  try {
    const os = require("os"), pr = os.constants.priority;
    const pids = [process.pid];
    try { app.getAppMetrics().forEach(x => { if (x.pid && pids.indexOf(x.pid) < 0) pids.push(x.pid); }); } catch (_) {}
    pids.forEach(id => { try { os.setPriority(id, level === "low" ? pr.PRIORITY_BELOW_NORMAL : pr.PRIORITY_NORMAL); } catch (_) {} });
  } catch (_) {}
}
function gmGamePrio(exe, high) {
  if (!/^[\w .()-]{1,80}$/.test(String(exe || ""))) return;
  const ps = "Get-Process -Name '" + String(exe).replace(/'/g, "") + "' -ErrorAction SilentlyContinue | ForEach-Object { try { $_.PriorityClass = '" + (high ? "AboveNormal" : "Normal") + "' } catch {} }";
  try { execFile("powershell.exe", ["-NoProfile", "-NonInteractive", "-WindowStyle", "Hidden", "-Command", ps], { windowsHide: true, timeout: 8000 }, () => {}); } catch (_) {}
}

  if (want) {

const pmExe = () => app.isPackaged ? path.join(process.resourcesPath, "PresentMon.exe") : path.join(__dirname, "bin", "PresentMon.exe");

const pm = { p: null, exe: "", ok: false, err: "", got: 0, hdr: null, buf: "", frames: [], tries: 0, triedFor: "", warned: false, timer: null, last: 0 };
function pmWanted() { return !!(!gmOn && S.enabled && S.live && S.game && S.fpsreal && gameState.full && gameState.exe); }
function pmStop() {
  if (pm.timer) { clearInterval(pm.timer); pm.timer = null; }
  const p = pm.p; pm.p = null; pm.ok = false; pm.got = 0; pm.hdr = null; pm.buf = ""; pm.frames = []; pm.exe = "";

function pmSync() {
  if (!pmWanted()) { if (pm.p || pm.timer) pmStop(); return; }
  const name = gameState.exe + ".exe";
  if (pm.p && pm.exe === name) return;
  pmStop();
  if (!pmExists()) { pm.err = "PresentMon tidak ada"; return; }
  if (!/^[\w .()-]{1,60}$/.test(name)) return;
  if (pm.triedFor !== name) { pm.triedFor = name; pm.tries = 0; }
  if (pm.tries >= 3) return;
  let p = null;
  try {
    p = spawn(pmExe(), ["-process_name", name, "-output_stdout", "-stop_existing_session", "-dont_restart_as_admin", "-no_top", "-terminate_on_proc_exit"], { windowsHide: true, stdio: ["ignore", "pipe", "pipe"] });
  } catch (e) { pm.err = String((e && e.message) || e); return; }
  pm.p = p; pm.exe = name; pm.err = ""; pm.got = 0; pm.hdr = null; pm.buf = ""; pm.frames = [];
  p.stdout.on("data", d => pmData(String(d)));
  p.stderr.on("data", d => { pm.err = (pm.err + String(d)).slice(-300); });
  p.on("error", e => { if (pm.p === p) pm.err = String((e && e.message) || e); });
  p.on("exit", () => {
    if (pm.p !== p) return;
    const denied = !pm.got && /access|administrator|privilege|Performance Log|elevat|denied/i.test(pm.err);
    pm.p = null; pm.ok = false;
    if (pm.timer) { clearInterval(pm.timer); pm.timer = null; }
    if (gameState.real) { gameState.real = false; gameState.fps = -1; gameState.low = -1; refreshGame(); }
    if (!pm.got) {
      pm.tries++;
      if (denied && !pm.warned && S.enabled && !quiet()) {
        pm.warned = true;
        showIsland({ title: T("FPS akurat butuh izin"), text: T('Buka pengaturan lalu tekan "Beri izin FPS" (sekali saja)'), app: "FPS", icon: svgIcon("#ff9f0a", "\u26A0"), appId: "", quiet: true });
      }
      if (!denied) setTimeout(pmSync, 4000);
    } else setTimeout(pmSync, 2000);
  });
  pm.timer = setInterval(pmTick, 500);
}
function pmData(chunk) {
  pm.buf += chunk;
  if (pm.buf.length > 200000) pm.buf = pm.buf.slice(-20000);
  let i;
  while ((i = pm.buf.indexOf("\n")) >= 0) {
    const line = pm.buf.slice(0, i).replace(/\r$/, ""); pm.buf = pm.buf.slice(i + 1);
    if (!line) continue;
    const c = line.split(",");
    if (!pm.hdr) {
      if (/Application/i.test(line) && /msBetweenPresents|FrameTime/i.test(line)) {
        let ft = c.indexOf("msBetweenPresents"); if (ft < 0) ft = c.indexOf("FrameTime");
        pm.hdr = { ft: ft };
      }
      continue;
    }
    const ms = parseFloat(c[pm.hdr.ft]);
    if (isFinite(ms) && ms > 0 && ms < 5000) { pm.frames.push(ms); pm.got++; pm.last = Date.now(); if (pm.frames.length > 900) pm.frames.splice(0, pm.frames.length - 900); }
  }
}
function pmTick() {
  if (!pm.p) return;
  const f = pm.frames;
  if (!f.length) return;
  if (Date.now() - pm.last > 2500) { gameState.fps = 0; gameState.low = 0; gameState.real = true; refreshGame(); return; }
  let acc = 0, n = 0;
  for (let k = f.length - 1; k >= 0 && acc < 1000; k--) { acc += f[k]; n++; }
  const fps = acc > 0 ? Math.round(n * 1000 / acc) : -1;
  const w = f.slice(-300).sort((a, b) => b - a), cnt = Math.max(3, Math.ceil(w.length * 0.01));
  let sum = 0; for (let k = 0; k < cnt && k < w.length; k++) sum += w[k];
  const low = sum > 0 ? Math.round(1000 / (sum / Math.min(cnt, w.length))) : -1;
  pm.ok = true; gameState.fps = fps; gameState.low = low; gameState.real = true;
  refreshGame();
}

const baseOf = p => String(p || "").replace(/^.*[\\\/]/, "").toLowerCase();
const isDrivePath = s => /^[a-z]:[\\\/]/i.test(String(s || ""));
function lnkTarget(p) {
  if (lnkCache.has(p)) return lnkCache.get(p);
  let t = "";
  try { t = String(shell.readShortcutLink(p).target || ""); } catch (_) {}
  lnkCache.set(p, t);
  return t;
}
function pinMatches(pin, w) {
  const id = String(pin.id || "").toLowerCase();
  if (!id) return false;
  const x = String(w.x || "").toLowerCase(), a = String(w.a || "").toLowerCase();
  if (a && id === a) return true;
  if (x && id === x) return true;
  if (/\.lnk$/.test(id) && x) { const t = lnkTarget(pin.id).toLowerCase(); if (t && t === x) return true; }
  if (!isDrivePath(id) && /\.exe$/.test(id) && x && baseOf(id) === baseOf(x)) return true;   // AppID gaya "{GUID}\...\app.exe"
  return false;
}
function dockNameFor(x, a) {
  if (a) return prettyApp(a);
  const b = baseOf(x);
  return b ? niceName(b) : "App";
}
/* ----- badge notifikasi di ikon Dock ----- */
const dockBadges = new Map();   // id aplikasi (huruf kecil) -> { id, c } jumlah notifikasi yang belum "dilihat"
const normName = s => String(s || "").toLowerCase().replace(/\.exe$/, "").replace(/[^a-z0-9]/g, "");
function badgeHits(k, v, id, name, ws) {
  if (id && String(id).toLowerCase() === k) return true;
  if (ws.some(w => pinMatches({ id: v.id }, w))) return true;
  const a = normName(name), b = normName(prettyApp(v.id));
  return !!a && a === b;
}
function dockBadgeAdd(appId) {
  if (S.dbadge === false || !S.dock || !appId) return;
  const id = String(appId), k = id.toLowerCase();
  const fgw = dockWins.find(w => w.h === dockFg);
  if (fgw && badgeHits(k, { id: id }, fgw.a || fgw.x, dockNameFor(fgw.x, fgw.a), [fgw])) return;   // aplikasinya sedang dilihat
  const v = dockBadges.get(k) || { id: id, c: 0 };
  v.c = Math.min(999, v.c + 1); dockBadges.set(k, v);
  dockPush(false);
}
function dockBadgeCount(id, name, ws) {
  let n = 0;
  dockBadges.forEach((v, k) => { if (badgeHits(k, v, id, name, ws)) n += v.c; });
  return n;
}
function dockBadgeClear(id, name, ws) {
  let ch = false;
  Array.from(dockBadges.keys()).forEach(k => { if (badgeHits(k, dockBadges.get(k), id, name, ws)) { dockBadges.delete(k); ch = true; } });
  return ch;
}
function dockBadgeClearFg() {
  const w = dockWins.find(x => x.h === dockFg);
  if (w) dockBadgeClear(w.a || w.x, dockNameFor(w.x, w.a), [w]);
}
/* ----- Adaptive Dock: urutan & saran mengikuti aplikasi yang paling sering dipakai ----- */
const DUSE_HALF = 7 * 864e5;
function dockScore(u) { return u ? u.n * Math.pow(0.5, Math.max(0, Date.now() - u.t) / DUSE_HALF) : 0; }
function dockUse(id, name) {
  if (S.dadapt === false || !id) return;
  id = String(id); const k = id.toLowerCase();
  if (!S.duse || typeof S.duse !== "object") S.duse = {};
  const u = S.duse[k];
  S.duse[k] = { id: id, name: String(name || (u && u.name) || prettyApp(id)).slice(0, 60), n: dockScore(u) + 1, t: Date.now() };
  const ks = Object.keys(S.duse);
  if (ks.length > 60) { ks.sort((a, b) => dockScore(S.duse[a]) - dockScore(S.duse[b])); ks.slice(0, ks.length - 60).forEach(x => delete S.duse[x]); }
  save();
}
let dockUseFg = 0, dockRankAt = 0; const dockRank = new Map();
function dockUseFgTrack() {
  if (!dockFg || dockFg === dockUseFg) return;
  dockUseFg = dockFg;
  const w = dockWins.find(x => x.h === dockFg);
  const id = w && (w.a || w.x);
  if (id) dockUse(id, dockNameFor(w.x, w.a));
}
function dockBuild() {
  const groups = new Map();
  dockWins.forEach(w => {
    const k = w.a ? "a:" + String(w.a).toLowerCase() : "x:" + String(w.x || "").toLowerCase();
    let g = groups.get(k);
    if (!g) { g = { k: k, ws: [], x: w.x, a: w.a }; groups.set(k, g); }
    g.ws.push(w);
  });
  const used = new Set(), map = new Map(), pins = [], runs = [];
  groups.forEach((g, k) => { if (!dockOrd.has("r:" + k)) dockOrd.set("r:" + k, ++dockSeq); });
  (Array.isArray(S.dpins) ? S.dpins : []).forEach(p => {
    let ws = [];
    groups.forEach((g, k) => { if (!used.has(k) && g.ws.some(w => pinMatches(p, w))) { used.add(k); ws = ws.concat(g.ws); } });
    const it = { k: "p:" + String(p.id).toLowerCase(), t: "app", name: String(p.name || prettyApp(p.id)), run: ws.length, act: ws.some(w => w.h === dockFg), pin: true, ic: String(p.id) };
    const bd = S.dbadge !== false ? dockBadgeCount(p.id, it.name, ws) : 0; if (bd) it.bd = bd; map.set(it.k, { id: p.id, name: it.name, ws: ws, pin: true });
    pins.push(it);
  });
  if (S.drun !== false) {
    groups.forEach((g, k) => {
      if (used.has(k)) return;
      const id = g.a || g.x;
      if (!id) return;
      const it = { k: "r:" + k, t: "app", name: dockNameFor(g.x, g.a), run: g.ws.length, act: g.ws.some(w => w.h === dockFg), pin: false, ic: String(id) };
      const bd = S.dbadge !== false ? dockBadgeCount(id, it.name, g.ws) : 0; if (bd) it.bd = bd; map.set(it.k, { id: id, name: it.name, ws: g.ws, pin: false });
      it.o = dockOrd.get(it.k); runs.push(it);
    });
    if (S.dadapt !== false) {
      const sc = it => dockScore((S.duse || {})[String(map.get(it.k).id).toLowerCase()]);
      // urutan dibekukan (dihitung ulang tiap 5 menit atau saat ada aplikasi baru) supaya ikon tidak lompat-lompat saat dipakai
      if (Date.now() - dockRankAt > 300000 || runs.some(r => !dockRank.has(r.k))) {
        dockRank.clear(); dockRankAt = Date.now();
        runs.slice().sort((a, b) => (sc(b) - sc(a)) || ((a.o | 0) - (b.o | 0))).forEach((r, i) => dockRank.set(r.k, i));
      }
      runs.sort((a, b) => (dockRank.get(a.k) | 0) - (dockRank.get(b.k) | 0));
    } else runs.sort((a, b) => (a.o | 0) - (b.o | 0));
  }
  if (S.dadapt !== false && S.drun !== false) {                     // saran: aplikasi yang sering dipakai tapi sedang tidak jalan (maks 3)
    const have = new Set();
    map.forEach(v => have.add(String(v.id).toLowerCase()));
    Object.keys(S.duse || {}).map(k => ({ k: k, u: S.duse[k], s: dockScore(S.duse[k]) }))
      .filter(x => x.s >= 3 && !have.has(x.k)).sort((a, b) => b.s - a.s).slice(0, 3)
      .forEach(x => {
        const it = { k: "f:" + x.k, t: "app", name: x.u.name, run: 0, act: false, pin: false, ic: String(x.u.id), fav: true };
        map.set(it.k, { id: x.u.id, name: x.u.name, ws: [], pin: false });
        runs.push(it);
      });
  }
  const items = [{ k: "st", t: "st", name: "" }].concat(S.lp !== false ? [{ k: "lp", t: "lp", name: "" }] : [], S.mc !== false ? [{ k: "mc", t: "mc", name: "" }] : [], pins);
  if (pins.length && runs.length) items.push({ k: "s1", t: "sep" });
  runs.forEach(r => items.push(r));
  items.push({ k: "s2", t: "sep" });
  items.push({ k: "dl", t: "dl", name: "" });
  items.push({ k: "bin", t: "bin", name: "" });
  if (S.dtb) items.push({ k: "tb", t: "tb", name: "" });
  items.push({ k: "ic", t: "ic", name: "", hid: !!S.dicons });
  return { items: items, map: map };
}
function dockPush(force) {
  if (!dockReady) return;
  const b = dockBuild();
  dockMap = b.map;
  const sig = JSON.stringify(b.items);
  if (!force && sig === dockSig) return;
  dockSig = sig;
  dockSend("items", b.items);
  b.items.forEach(it => { if (it.ic) dockIcon(it.ic); });
}
function dockIcon(key) {
  if (dockIconData.has(key) || dockIconBusy.has(key)) return;
  if (Date.now() - (dockIconFail.get(key) || 0) < 60000) return;
  dockIconBusy.add(key);
  appIcon(key, 128)
    .then(v => v || (isDrivePath(key) && !/\.lnk$/i.test(key) ? exeIcon(key) : ""))
    .then(v => {
      dockIconBusy.delete(key);
      if (v) { dockIconData.set(key, v); const o = {}; o[key] = v; dockSend("icons", o); }
      else dockIconFail.set(key, Date.now());
    })
    .catch(() => { dockIconBusy.delete(key); dockIconFail.set(key, Date.now()); });
}

// Setelah aksi dari Dock (buka/pulihkan/fokus), jendela lain naik ke depan: pasang ulang status overlay Dock supaya tetap bisa diklik.
function dockResync() {
  [120, 450, 1100].forEach(ms => setTimeout(() => {
    try {
      if (!S.dock || !dockWin || dockWin.isDestroyed() || gmOn) return;
      placeDock(); dockVis();
      if (dockWin.isVisible() && dockIgn) { dockWin.setAlwaysOnTop(true, "screen-saver"); dockWin.moveTop(); }   // kursor sedang di Dock: jangan ganggu klik
      dockIgn = true; dockAssert = 0; dockCurKey = ""; dockWin.setIgnoreMouseEvents(true, { forward: true });
      clearTimeout(dockT); dockTick();                       // dinilai ulang SEKARANG, tanpa jeda yang membuat klik hilang
    } catch (_) {}
  }, ms));
}
function dockCmd(l) { if (!dockProc) return; if (/^(focus|show|min|close|max|place)\b/.test(l)) dockResync(); try { dockProc.stdin.write(l + "\n"); } catch (_) {} }
function onDockMsg(m) {
  if (m && m.t === "mv") { try { snapMv(m); } catch (_) {} return; }
  if (m && m.t === "shown") {
    clearTimeout(dockPendT); const id = dockPendId; dockPendId = "";
    if (id && !m.ok) openApp(id);
    return;
  }
  if (!m || m.t !== "wins") return;
  dockRestarts = 0; dockErr = "";
  dockWins = Array.isArray(m.l) ? m.l.filter(w => w && Number.isInteger(w.h) && typeof w.x === "string") : [];
  dockFg = Number(m.fg) || 0;
  try { dockUseFgTrack(); } catch (_) {}
  if (dockBadges.size) dockBadgeClearFg();
  const fsNow = !!m.fs;
  if (fsNow !== dockFs) { dockFs = fsNow; dockVis(); }
  dockPush(false);
}
function startDockProc() {
  if (gmOn) return;
  if (dockProc) return;
  if (!dockExists()) { dockErr = "fxdock.exe tidak ditemukan"; return; }
  dockErr = "";
  dockProc = runHelper(["watch", String(process.pid), tbFlag(), icFlag()], onDockMsg, () => {
    dockProc = null; dockWins = [];
    if (S.dock && !quitting) {
      dockPush(false);
      if (dockRestarts < 6) { dockRestarts++; setTimeout(() => { if (S.dock && !quitting) startDockProc(); }, 1500); }
      else dockErr = "pembantu berhenti";
    }
  }, dockExe());
}
function stopDockProc() { const p = dockProc; dockProc = null; killHelper(p); dockWins = []; }

const tbFlag = () => path.join(app.getPath("userData"), "dock-taskbar.json");
function tbRun(args) {
  try { return String(require("child_process").execFileSync(dockExe(), args, { windowsHide: true, timeout: 5000, encoding: "utf8" })).trim(); } catch (_) { return ""; }
}
function tbOrig() {
  try { const j = JSON.parse(fs.readFileSync(tbFlag(), "utf8")); return j && (j.orig === 0 || j.orig === 1) ? j.orig : -1; } catch (_) { return -1; }
}
let tbShown = false;
function tbHide() {
  if (!dockExists()) return;
  let orig = tbOrig();
  if (orig < 0) {
    const cur = tbRun(["tb", "get"]);
    if (cur !== "0" && cur !== "1") return;
    orig = Number(cur);
    try { fs.writeFileSync(tbFlag(), JSON.stringify({ orig: orig })); } catch (_) {}   // dicatat dulu: kalau aplikasi mati mendadak, start berikutnya mengembalikannya
  }
  tbRun(["tb", "set", "1"]);       // bebaskan ruang kerja
  tbRun(["tb", "vis", "0"]);       // lalu sembunyikan total: tidak muncul lagi walau kursor ke tepi bawah
  tbShown = false;
}
// Tombol "munculkan taskbar": bolak-balik antara tampil penuh dan tersembunyi total.
function tbToggle() {
  if (!S.dock || !S.dtb || tbOrig() < 0) return;
  if (tbShown) { tbHide(); return; }
  tbRun(["tb", "vis", "1"]);
  tbRun(["tb", "set", String(tbOrig())]);
  tbShown = true;
}
function tbRestore() {
  const orig = tbOrig();
  if (orig < 0) return;
  tbRun(["tb", "vis", "1"]);
  tbRun(["tb", "set", String(orig)]);
  try { fs.unlinkSync(tbFlag()); } catch (_) {}
  tbShown = false;
}
function tbKeys(on) {
  try { globalShortcut.unregister("CommandOrControl+Alt+T"); globalShortcut.unregister("CommandOrControl+Alt+D"); } catch (_) {}
  if (on) { try { globalShortcut.register("CommandOrControl+Alt+T", () => tbToggle()); } catch (_) {} try { globalShortcut.register("CommandOrControl+Alt+D", () => setOpt("dicons", !S.dicons)); } catch (_) {} }
}

/* ----- klik-tembus & kursor ----- */
function dockTick() {
  dockT = null;
  if (!S.dock || !dockWin || dockWin.isDestroyed()) return;
  let next = 140;
  try {
    if (dockReady && dockWin.isVisible()) {
      const pt = screen.getCursorScreenPoint(), b = dockWin.getBounds(), d = screen.getPrimaryDisplay().bounds;
      const x = pt.x - b.x, y = pt.y - b.y;
      const inb = x >= 0 && y >= 0 && x < b.width && y < b.height;
      const edge = !!S.dhide && pt.y >= d.y + d.height - 6 && pt.y < d.y + d.height && pt.x >= b.x && pt.x < b.x + b.width;
      let hit = false;
      if (inb) hit = dockRects.some(r => x >= r.x && x <= r.x + r.w && y >= r.y && y <= r.y + r.h);
      if (hit === dockIgn) { dockIgn = !hit; dockAssert = 0; dockWin.setIgnoreMouseEvents(!hit, { forward: true }); }
      else if (hit && ++dockAssert >= 30) { dockAssert = 0; dockWin.setIgnoreMouseEvents(false); }
      const key = (inb ? x + "," + y : "-") + (edge ? "e" : "");

      next = (inb || edge) ? 16 : 50;
    }
  } catch (_) {}
  dockT = setTimeout(dockTick, next);
}

}

function dockSeed() {
  const env = process.env, out = [];
  const add = (p, n) => { try { if (p && fs.existsSync(p)) { out.push({ id: p, name: n }); return true; } } catch (_) {} return false; };
  add(path.join(env.WINDIR || "C:\\Windows", "explorer.exe"), "File Explorer");
  add(path.join(env["ProgramFiles(x86)"] || "C:\\Program Files (x86)", "Microsoft", "Edge", "Application", "msedge.exe"), "Microsoft Edge");
  [path.join(env.ProgramFiles || "C:\\Program Files", "Google", "Chrome", "Application", "chrome.exe"),
   path.join(env["ProgramFiles(x86)"] || "C:\\Program Files (x86)", "Google", "Chrome", "Application", "chrome.exe"),
   path.join(env.LOCALAPPDATA || "", "Google", "Chrome", "Application", "chrome.exe")].some(p => add(p, "Google Chrome"));
  add(path.join(env.ProgramFiles || "C:\\Program Files", "Mozilla Firefox", "firefox.exe"), "Firefox");
  return out;
}
// Ubah id dari daftar Start Menu menjadi id yang bisa dicocokkan dengan jendela yang berjalan.
function resolvePinId(id, name) {
  id = String(id || "");
  if (isDrivePath(id)) {
    if (/\.lnk$/i.test(id)) { const t = lnkTarget(id); if (t && /\.exe$/i.test(t) && fs.existsSync(t)) return t; }
    return id;
  }
  if (id.indexOf("!") > 0 || /\.exe$/i.test(id)) return id;      // aplikasi Store (cocok lewat AUMID) atau "{GUID}\...\app.exe" (cocok lewat nama exe)
  try {                                                         // AppID klasik, mis. "Chrome": cari pintasan Start Menu bernama sama
    const list = [];
    scanStartMenu(list, new Set());
    const n = String(name || appNames.get(id) || "").toLowerCase();
    const f = n && list.find(a => a.n.toLowerCase() === n);
    if (f) { const t = lnkTarget(f.p); if (t && /\.exe$/i.test(t) && fs.existsSync(t)) return t; }
  } catch (_) {}
  return id;
}
function dockSavePins() { save(); dockPush(true); }
function dockPinAdd(id, name) {
  id = String(id || "").slice(0, 400);
  if (!id) return;
  const rid = resolvePinId(id, name);
  if (!Array.isArray(S.dpins)) S.dpins = [];
  if (S.dpins.length >= 30 || S.dpins.some(p => String(p.id).toLowerCase() === rid.toLowerCase())) return;
  S.dpins.push({ id: rid, name: String(name || prettyApp(rid)).slice(0, 60) });
  dockSavePins();
}
function dockPinRemove(id) {
  id = String(id || "").toLowerCase();
  S.dpins = (S.dpins || []).filter(p => String(p.id).toLowerCase() !== id);
  dockSavePins();
}
function dockPinMove(id, dir) {

  const a = S.dpins || [], i = a.findIndex(p => String(p.id).toLowerCase() === id), j = i + (Number(dir) < 0 ? -1 : 1);
  if (i < 0 || j < 0 || j >= a.length) return;
  const t = a[i]; a[i] = a[j]; a[j] = t;
  dockSavePins();
}

let dockPendId = "", dockPendT = null;
function exeBaseOf(id) {
  id = String(id || "");
  if (/\.lnk$/i.test(id) && isDrivePath(id)) id = lnkTarget(id) || "";
  return /\.exe$/i.test(id) ? baseOf(id) : "";
}
function dockShowOrOpen(it) {
  const eb = exeBaseOf(it.id);
  if (!eb || !dockProc || !/^[\w .()-]{1,80}\.exe$/i.test(eb)) { openApp(it.id); return; }
  dockPendId = String(it.id);
  clearTimeout(dockPendT);
  dockPendT = setTimeout(() => { const id = dockPendId; dockPendId = ""; if (id) openApp(id); }, 1500);   // pembantu lama / tidak menjawab
  dockCmd("show " + eb);
}
function dockClick(it) {
  const ws = it.ws;
  if (dockBadgeClear(it.id, it.name, ws)) dockPush(false);
  if (!ws.length) { dockShowOrOpen(it); return; }
  const fi = ws.findIndex(w => w.h === dockFg && !w.m);                          // aktif dan tidak sedang diminimize
  if (fi >= 0) {
    if (ws.length === 1) dockCmd("focus " + ws[0].h);                            // aplikasi aktif: tetap tampil (tidak diminimize)
    else dockCmd("focus " + ws[(fi + 1) % ws.length].h);                          // banyak jendela: berpindah ke jendela berikutnya
  } else {
    const t = ws.find(w => !w.m) || ws[0];                                       // yang diminimize dipulihkan
    dockCmd("focus " + t.h);
  }
}
function dockAct(p) {
  if (!p || typeof p.a !== "string" || typeof p.k !== "string") return;
  if (p.k === "tb") { tbToggle(); return; }
  if (p.k === "ic") { setOpt("dicons", !S.dicons); return; }
  if (p.k === "st") { dockCmd("start"); return; }
            if (p.k === "lp") { toggleLp(); return; }
  if (p.k === "mc") { toggleMc(); return; }
  if (p.k === "dl") { try { shell.openPath(app.getPath("downloads")); } catch (_) {} return; }
  if (p.k === "bin") { try { spawn("explorer.exe", ["shell:RecycleBinFolder"], { detached: true, stdio: "ignore", windowsHide: true }).unref(); } catch (_) {} return; }
  const it = dockMap.get(p.k);
  if (!it) return;
  if (p.a === "click") { dockUse(it.id, it.name); dockClick(it); }
  else if (p.a === "launch") { dockUse(it.id, it.name); openApp(it.id); }
  else if (p.a === "quit") it.ws.forEach(w => dockCmd("close " + w.h));
  else if (p.a === "pin") { if (!it.pin) dockPinAdd(it.id, it.name); }
  else if (p.a === "unpin") { if (it.pin) dockPinRemove(it.id); }
}

function snapGeom(d) {   // area strip dan posisi tiap kartu (DIP)
  const wa = d.workArea, total = SNAP_N * SNAP_CH + (SNAP_N - 1) * SNAP_GAP;
  const x = d.bounds.x + d.bounds.width - SNAP_W, top = wa.y + Math.max(0, Math.round((wa.height - total) / 2));
  return { x: x, y: wa.y, w: SNAP_W, h: wa.height, top: top, cx0: x + (SNAP_W - 84) / 2, cx1: x + (SNAP_W + 84) / 2 };
}
function snapIdx(pt, g) {
  if (pt.x < g.cx0 || pt.x > g.cx1) return -1;
  const rel = pt.y - g.top, i = Math.floor(rel / (SNAP_CH + SNAP_GAP));
  return (i >= 0 && i < SNAP_N && rel - i * (SNAP_CH + SNAP_GAP) <= SNAP_CH) ? i : -1;
}

async function spotBuild() {
  if (spotIdxBusy) return;
  spotIdxBusy = true;
  const out = [], seen = new Set(), q = spotRoots().map(r => [r, 0]);
  try {
    while (q.length && out.length < 60000) {
      const [d, dep] = q.shift();
      let es = [];
      try { es = await fs.promises.readdir(d, { withFileTypes: true }); } catch (_) { continue; }
      for (const e of es) {
        if (e.name[0] === "." || e.name[0] === "~" || e.name === "node_modules" || /^desktop\.ini$/i.test(e.name)) continue;
        const f = path.join(d, e.name);
        if (seen.has(f)) continue;
        seen.add(f);
        if (e.isDirectory()) { if (dep < 4) q.push([f, dep + 1]); out.push({ n: e.name, p: f, d: 1 }); }
        else out.push({ n: e.name, p: f, d: 0 });
      }
    }
  } catch (_) {}
  spotIdx = out; spotIdxAt = Date.now(); spotIdxBusy = false;
}
function spotScore(name, toks) {
  const n = String(name).toLowerCase();
  let s = 0;
  for (const t of toks) { const i = n.indexOf(t); if (i < 0) return -1; s += i === 0 ? 0 : (/[\s._-]/.test(n[i - 1]) ? 1 : 3); }
  return s * 1000 + n.length;
}
function spotCalc(q) {
  const FN = ["sqrt", "sin", "cos", "tan", "log", "ln", "abs", "round", "floor", "ceil"], MAP = ["sqrt", "sin", "cos", "tan", "log10", "log", "abs", "round", "floor", "ceil"];
  let s = String(q).toLowerCase().replace(/,/g, ".").replace(/\^/g, "**").replace(/\u00d7/g, "*").replace(/\u00f7/g, "/").replace(/\bpi\b/g, "(3.141592653589793)");
  s = s.replace(/\b(sqrt|sin|cos|tan|log|ln|abs|round|floor|ceil)\b/g, m => "#" + FN.indexOf(m));
  if (!/^[0-9+\-*\/%().\s#]+$/.test(s) || !/\d/.test(s) || !/[+*\/%#]|[\d)]\s*-/.test(s)) return null;
  s = s.replace(/#(\d)/g, (_m, k) => "Math." + MAP[+k]);
  try {
    const v = Function('"use strict"; return (' + s + ")")();
    if (typeof v !== "number" || !isFinite(v)) return null;
    const raw = String(Number(v.toPrecision(12)));
    let shown = raw;
    try { shown = v.toLocaleString(I18N.loc(S.lang), { maximumFractionDigits: 10 }); } catch (_) {}
    return { text: shown, raw: raw };
  } catch (_) { return null; }
}
function spotSearch(qs) {

  const c = spotCalc(q);
  if (c) res.push({ t: "calc", n: "= " + c.text, s: q + "  \u00b7 " + T("Enter untuk menyalin"), raw: c.raw });
  const toks = q.toLowerCase().split(/\s+/).filter(Boolean);
  startApps.map(a => ({ a: a, sc: spotScore(a.n, toks) })).filter(x => x.sc >= 0).sort((x, y) => x.sc - y.sc).slice(0, 6)
    .forEach(x => res.push({ t: "app", n: x.a.n, s: T("Aplikasi"), p: x.a.p }));
  if (S.spotfiles !== false && q.length >= 2) {
    spotIdx.map(f => ({ f: f, sc: spotScore(f.n, toks) })).filter(x => x.sc >= 0).sort((x, y) => x.sc - y.sc).slice(0, 6)
      .forEach(x => res.push({ t: x.f.d ? "dir" : "file", n: x.f.n, s: path.dirname(x.f.p).replace(require("os").homedir(), "~"), p: x.f.p }));

  return res.slice(0, 12);
}

const ccSend = (ch, d) => { if (ccWin && !ccWin.isDestroyed() && ccReady) { try { ccWin.webContents.send("cc:" + ch, d); } catch (_) {} } };

function ccReadDark(cb) {
  execFile("reg", ["query", CC_KEY, "/v", "AppsUseLightTheme"], { windowsHide: true, timeout: 3000 }, (_e, out) => {
    const m = /AppsUseLightTheme\s+REG_DWORD\s+0x([0-9a-f]+)/i.exec(String(out || ""));
    cb(m ? parseInt(m[1], 16) === 0 : false);
  });
}
function ccSetDark(on) {
  const v = on ? "0" : "1";
  ["AppsUseLightTheme", "SystemUsesLightTheme"].forEach(k => {
    try { execFile("reg", ["add", CC_KEY, "/v", k, "/t", "REG_DWORD", "/d", v, "/f"], { windowsHide: true, timeout: 3000 }, () => { sysCmd("theme"); }); } catch (_) {}
  });
  ccDark = !!on; ccSend("dark", ccDark);
}
function ccPs(script, cb) {
  try {
    execFile("powershell.exe", ["-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass", "-Command", script], { windowsHide: true, timeout: 8000 },
      (e, out) => cb(e, String(out || "").trim()));
  } catch (_) { cb(true, ""); }
}
function ccReadBri() {
  ccPs("try { (Get-CimInstance -Namespace root/WMI -ClassName WmiMonitorBrightness -ErrorAction Stop | Select-Object -First 1).CurrentBrightness } catch { '' }", (_e, o) => {
    const n = parseInt(o, 10);
    ccBri = isFinite(n) ? Math.max(0, Math.min(100, n)) : -1;
    ccSend("bri", ccBri);
  });
}
function ccSetBri(v) {
  v = Math.round(clamp(v, 5, 100)); ccBri = v;
  clearTimeout(ccBriT);
  ccBriT = setTimeout(() => ccPs("Get-CimInstance -Namespace root/WMI -ClassName WmiMonitorBrightnessMethods | Invoke-CimMethod -MethodName WmiSetBrightness -Arguments @{Timeout=1;Brightness=" + v + "} | Out-Null", () => {}), 150);
}

function ccWifiToggle() {
  const ps = "Add-Type -AssemblyName System.Runtime.WindowsRuntime;"
    + "[Windows.Devices.Radios.Radio,Windows.System.Devices,ContentType=WindowsRuntime]|Out-Null;"
    + "$as=([System.WindowsRuntimeSystemExtensions].GetMethods()|?{$_.Name -eq 'AsTask' -and $_.GetParameters().Count -eq 1 -and $_.GetParameters()[0].ParameterType.Name -eq 'IAsyncOperation`1'})[0];"
    + "function W($o,$t){$m=$as.MakeGenericMethod($t);$k=$m.Invoke($null,@($o));$k.Wait(-1)|Out-Null;$k.Result};"
    + "$rs=W ([Windows.Devices.Radios.Radio]::GetRadiosAsync()) ([System.Collections.Generic.IReadOnlyList[Windows.Devices.Radios.Radio]]);"
    + "foreach($r in $rs){if($r.Kind -eq 'WiFi'){$n=if($r.State -eq 'On'){'Off'}else{'On'};"
    + "$x=$r.SetStateAsync($n);$k=$as.MakeGenericMethod([Windows.Devices.Radios.RadioAccessStatus]).Invoke($null,@($x));$k.Wait(-1)|Out-Null}}";
  try {
    const enc = Buffer.from(ps, "utf16le").toString("base64");
    require("child_process").execFile("powershell.exe", ["-NoProfile", "-NonInteractive", "-WindowStyle", "Hidden", "-EncodedCommand", enc], { windowsHide: true, timeout: 15000 }, () => {});
  } catch (_) {}
}
function ccAct(p) {
  if (p.a === "dark") ccSetDark(!ccDark);
  else if (p.a === "dnd") { S.dnd = !S.dnd; save(); ccSend("dnd", !!S.dnd); }
  else if (p.a === "vol") sysCmd("vol " + Math.round(clamp(p.v, 0, 100)));
  else if (p.a === "mute") sysCmd("mute");
  else if (p.a === "bri") ccSetBri(p.v);
  else if (p.a === "wifi") ccWifiToggle();
  else if (p.a === "hide") hideCc();
}

const mcHas = h => Number.isInteger(h) && mcList.some(w => w.h === h);
function mcSnap(cb) {
  if (!dockExists()) { cb(null); return; }
  try {
    execFile(dockExe(), ["snap"], { windowsHide: true, timeout: 5000, maxBuffer: 4 * 1024 * 1024, encoding: "utf8" }, (err, out) => {
      if (err) { cb(null); return; }
      let m = null;
      try { m = JSON.parse(String(out).trim().split("\n").pop()); } catch (_) {}
      cb(m && Array.isArray(m.l) ? m : null);
    });
  } catch (_) { cb(null); }
}
function mcCmd(l) {
  if (dockProc) { dockCmd(l); return; }
  if (!dockExists()) return;
  try { execFile(dockExe(), ["cmd"].concat(l.split(" ")), { windowsHide: true, timeout: 4000 }, () => {}); } catch (_) {}
}
// jendela milik overlay kita sendiri tidak ikut ditampilkan
function mcOwn() {
  const s = new Set();
  [mcWin, lpWin, ccWin, ncWin, spotWin, dockWin, mbWin, mbPop, isl].concat(M2 ? M2.wins() : []).forEach(w => {
    try {
      if (!w || w.isDestroyed()) return;
      const b = w.getNativeWindowHandle();
      s.add(String(b.length >= 8 ? Number(b.readBigUInt64LE(0)) : b.readUInt32LE(0)));
    } catch (_) {}
  });
  return s;
}
function mcName(x) {
  const n = String(x || "").replace(/^.*[\\/]/, "").replace(/\.exe$/i, "");
  return n ? n.charAt(0).toUpperCase() + n.slice(1) : "";
}
function mcThumbs(seq) {
  try {
    desktopCapturer.getSources({ types: ["window"], thumbnailSize: { width: 560, height: 350 }, fetchWindowIcons: false }).then(src => {
      if (seq !== mcSeq) return;
      const o = {};
      src.forEach(s => {
        const m = /^window:(\d+):/.exec(String(s.id));
        if (!m || !mcList.some(w => String(w.h) === m[1]) || !s.thumbnail || s.thumbnail.isEmpty()) return;
        const sz = s.thumbnail.getSize();
        if (sz.width < 8 || sz.height < 8) return;
        try { o[m[1]] = "data:image/jpeg;base64," + s.thumbnail.toJPEG(72).toString("base64"); } catch (_) {}
      });
      mcSend("thumbs", o);
    }).catch(() => {});
  } catch (_) {}
}
function mcIcons(seq, items) {
  const done = new Set();
  items.forEach(i => {
    if (!i.k || done.has(i.k)) return;
    done.add(i.k);
    const p = isDrivePath(i.k) ? exeIcon(i.k) : appIcon(i.a || i.k, 96);
    p.then(v => { if (v && seq === mcSeq) { const o = {}; o[i.k] = v; mcSend("icons", o); } }).catch(() => {});
  });
}

function showMc() {
  if (mcOutT) { clearTimeout(mcOutT); mcOutT = null; }
  const w = ensureMc();
  if (!mcReady) { mcWant = true; return; }
  mcWant = false;
  const seq = ++mcSeq;
  mcSnap(m => {   // daftar dibaca sebelum overlay tampil supaya jendelanya sendiri tidak ikut terbaca
    if (seq !== mcSeq || !mcWin || mcWin.isDestroyed()) return;
    const own = mcOwn(), fg = m ? Number(m.fg) || 0 : 0;
    mcList = m ? m.l.filter(x => x && Number.isInteger(x.h) && typeof x.x === "string" && !own.has(String(x.h))).slice(0, 60) : [];
    const items = mcList.map(x => ({ h: x.h, k: x.x, a: x.a || "", n: mcName(x.x), t: String(x.t || "").slice(0, 120), m: x.m ? 1 : 0, fg: x.h === fg ? 1 : 0 }));
    try {
      const d = screen.getDisplayNearestPoint(screen.getCursorScreenPoint()).bounds;
      w.setBounds({ x: d.x, y: d.y, width: d.width, height: d.height });
      w.setAlwaysOnTop(true, "screen-saver");
      mcShownAt = Date.now();
      w.setIgnoreMouseEvents(false);
      w.show(); w.moveTop(); w.focus(); w.webContents.focus();
    } catch (_) {}
    mcSend("show", { items: items.map(i => ({ h: i.h, k: i.k, n: i.n, t: i.t, m: i.m, fg: i.fg })), err: m ? "" : T(dockExists() ? "pembaca berhenti" : "fxdock.exe tidak ditemukan"),
                     L: { none: T("Tidak ada jendela terbuka"), min: T("Diminimalkan") } });
    if (items.length) { mcIcons(seq, items); mcThumbs(seq); }
  });
}

let hcT = null, hcCorner = "", hcSince = 0, hcArmed = true;
function hcRun(a) {
  if (a === "lp") toggleLp();
  else if (a === "mc") { if (S.mc !== false) toggleMc(); }
  else if (a === "cc") toggleCc();
  else if (a === "spot") { if (S.spot !== false) toggleSpot(); }
  else if (a === "nc") toggleNc();
  else if (a === "lock") mbRun("rundll32.exe", ["user32.dll,LockWorkStation"]);
  else if (M2 && a === "ss") M2.saverNow();
  else if (M2 && a === "qn") M2.noteToggle();
  else if (M2 && a === "sw") M2.switcher();
  else if (M2 && a === "shot") M2.shotBar();
  else if (M2 && a === "dsleep") M2.displaySleep();
}
function hcTick() {
  try {
    if (!(S.hc_tl || S.hc_tr || S.hc_bl || S.hc_br) || gameState.full) { hcCorner = ""; return; }
    const pt = screen.getCursorScreenPoint(), d = screen.getDisplayNearestPoint(pt).bounds;
    const L = pt.x <= d.x + 1, R = pt.x >= d.x + d.width - 2, Tp = pt.y <= d.y + 1, B = pt.y >= d.y + d.height - 2;
    const c = Tp && L ? "tl" : Tp && R ? "tr" : B && L ? "bl" : B && R ? "br" : "";
    if (!c) {
      const away = pt.x > d.x + 60 && pt.x < d.x + d.width - 60 || pt.y > d.y + 60 && pt.y < d.y + d.height - 60;
      if (away) hcArmed = true;
      hcCorner = ""; return;
    }
    if (c !== hcCorner) { hcCorner = c; hcSince = Date.now(); return; }
    if (hcArmed && Date.now() - hcSince >= 180 && S["hc_" + c]) { hcArmed = false; hcRun(S["hc_" + c]); }
  } catch (_) {}
}

function trusted(e) {
  try {
    if (!ctx || ctx.win.isDestroyed() || e.sender !== ctx.win.webContents) return false;
    const u = e.senderFrame ? e.senderFrame.url : "";
    if (/^file:/i.test(u)) return true;
    // mode URL: hanya halaman dari alamat (https) yang diatur saat build yang boleh memakai jembatan pengaturan
    const cu = ctx.cfg && ctx.cfg.url ? new URL(ctx.cfg.url) : null;
    return !!cu && cu.protocol === "https:" && new URL(u).origin === cu.origin;
  } catch (_) { return false; }
}
